recvmmsg(2)

الحزمة: man-pages

التراخيص: BSD-2-Clause, BSD-3-Clause, BSD-4.3

recvmmsg(2) دليل استدعاءات النظام recvmmsg(2)

الاسم

recvmmsg - استقبال رسائل متعددة من مقبس

المكتبة

مكتبة C القياسية (libc،  -lc)

الملخص

#define _GNU_SOURCE         /* انظر feature_test_macros(7) */
#include <sys/socket.h>
int recvmmsg(unsigned int n;
             int sockfd, struct mmsghdr msgvec[n], unsigned int n,
             int flags, struct timespec *timeout);

الوصف

استدعاء النظام recvmmsg() هو امتداد لـ recvmsg(2) يسمح للمستدعي باستقبال رسائل متعددة من مقبس باستخدام استدعاء نظام واحد. (هذا له فوائد أداء لبعض التطبيقات.) امتداد آخر لـ recvmsg(2) هو دعم مهلة لعملية الاستقبال.

الوسيط sockfd هو واصف الملف للمقبس الذي سيتم استقبال البيانات منه.

الوسيط msgvec هو مؤشر إلى مصفوفة من هياكل mmsghdr. حجم هذه المصفوفة محدد في n.

يتم تعريف هيكل mmsghdr في <sys/socket.h> كالتالي:


struct mmsghdr {
    struct msghdr msg_hdr;  /* رأس الرسالة */
    unsigned int  msg_len;  /* عدد البايتات المستلمة للرأس */
};

الحقل msg_hdr هو هيكل msghdr، كما هو موضح في recvmsg(2). الحقل msg_len هو عدد البايتات التي تم إرجاعها للرسالة في المدخل. هذا الحقل له نفس قيمة قيمة الإرجاع لاستدعاء recvmsg(2) واحد على الرأس.

الوسيط flags يحتوي على الأعلام مدمجة معًا (ORed). الأعلام هي نفسها كما هو موثق لـ recvmsg(2)، مع الإضافة التالية:

MSG_WAITFORONE (منذ Linux 2.6.34)
يقوم بتشغيل MSG_DONTWAIT بعد استلام الرسالة الأولى.

الوسيط timeout يشير إلى struct timespec (انظر clock_gettime(2)) يحدد مهلة (ثواني بالإضافة إلى نانوثواني) لعملية الاستقبال (ولكن انظر BUGS!). (سيتم تقريب هذه الفترة إلى أعلى حسب دقة ساعة النظام، وتتسبب تأخيرات جدولة النواة في أن تتجاوز فترة الحظر بشكل طفيف.) إذا كان timeout يساوي NULL، فإن العملية تمنع إلى أجل غير مسمى.

استدعاء recvmmsg() الذي يمنع يتوقف حتى يتم استقبال n رسالة أو حتى انتهاء المهلة. استدعاء غير مانع يقرأ أكبر عدد ممكن من الرسائل المتاحة (حتى الحد المحدد بواسطة n) ويعود فورًا.

عند العودة من recvmmsg()، يتم تحديث العناصر المتتالية لـ msgvec لتضمين معلومات حول كل رسالة تم استقبالها: يحتوي msg_len على حجم الرسالة المستلمة؛ يتم تحديث الحقول الفرعية لـ msg_hdr كما هو موضح في recvmsg(2). قيمة الإرجاع للاستدعاء تشير إلى عدد عناصر msgvec التي تم تحديثها.

قيمة الإرجاع

عند النجاح، يعيد recvmmsg() عدد الرسائل المستلمة في msgvec؛ عند حدوث خطأ، يتم إرجاع -1، ويتم تعيين errno للإشارة إلى الخطأ.

الأخطاء

الأخطاء كما هو الحال بالنسبة لـ recvmsg(2). بالإضافة إلى ذلك، يمكن أن يحدث الخطأ التالي:

EINVAL
timeout غير صالح.

انظر أيضًا BUGS.

المعايير

Linux.

التاريخ

Linux 2.6.33، glibc 2.12.

أخطاء

الوسيط timeout لا يعمل كما هو مقصود. يتم التحقق من المهلة فقط بعد استلام كل حزمة بيانات، لذلك إذا تم استلام ما يصل إلى n-1 حزمة بيانات قبل انتهاء المهلة، ولكن بعد ذلك لم يتم استلام أي حزم بيانات أخرى، فسيتم حظر الاستدعاء إلى الأبد.

إذا حدث خطأ بعد استلام رسالة واحدة على الأقل، ينجح الاستدعاء، ويعيد عدد الرسائل المستلمة. من المتوقع أن يتم إرجاع رمز الخطأ في استدعاء لاحق لـ recvmmsg(). في التنفيذ الحالي، ومع ذلك، يمكن أن يتم الكتابة فوق رمز الخطأ في هذه الأثناء بواسطة حدث شبكة غير ذي صلة على مقبس، على سبيل المثال حزمة ICMP واردة.

أمثلة

يستخدم البرنامج التالي recvmmsg() لاستقبال رسائل متعددة من مقبس وتخزينها في مخازن مؤقتة متعددة. يعود الاستدعاء إذا تم ملء جميع المخازن المؤقتة أو إذا انتهت المهلة المحددة.

يقوم المقتطف التالي بإنشاء حزم بيانات UDP بشكل دوري تحتوي على رقم عشوائي:


$ while true; do echo $RANDOM > /dev/udp/127.0.0.1/1234;
      sleep 0.25; done

يتم قراءة حزم البيانات هذه بواسطة التطبيق المثال، والذي يمكن أن يعطي الإخراج التالي:


$ ./a.out
5 رسائل تم استقبالها
1 11782
2 11345
3 304
4 13514
5 28421

مصدر البرنامج

#define _GNU_SOURCE
#include <arpa/inet.h>
#include <netinet/in.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/socket.h>
#include <time.h>
int
main(void)
{
#define VLEN 10
#define BUFSIZE 200
#define TIMEOUT 1
    int                 sockfd, retval;
    char                bufs[VLEN][BUFSIZE+1];
    struct iovec        iovecs[VLEN];
    struct mmsghdr      msgs[VLEN];
    struct timespec     timeout;
    struct sockaddr_in  addr;
    sockfd = socket(AF_INET, SOCK_DGRAM, 0);
    if (sockfd == -1) {
        perror("socket()");
        exit(EXIT_FAILURE);
    }
    addr.sin_family = AF_INET;
    addr.sin_addr.s_addr = htonl(INADDR_LOOPBACK);
    addr.sin_port = htons(1234);
    if (bind(sockfd, (struct sockaddr *) &addr, sizeof(addr)) == -1) {
        perror("bind()");
        exit(EXIT_FAILURE);
    }
    memset(msgs, 0, sizeof(msgs));
    for (size_t i = 0; i < VLEN; i++) {
        iovecs[i].iov_base         = bufs[i];
        iovecs[i].iov_len          = BUFSIZE;
        msgs[i].msg_hdr.msg_iov    = &iovecs[i];
        msgs[i].msg_hdr.msg_iovlen = 1;
    }
    timeout.tv_sec = TIMEOUT;
    timeout.tv_nsec = 0;
    retval = recvmmsg(sockfd, msgs, VLEN, 0, &timeout);
    if (retval == -1) {
        perror("recvmmsg()");
        exit(EXIT_FAILURE);
    }
    printf("%d messages received\n",retval);
    for (size_t i = 0; i < retval; i++) {
        bufs[i][msgs[i].msg_len] = 0;
        printf("%zu %s", i+1, bufs[i]);
    }
    exit(EXIT_SUCCESS);
}
