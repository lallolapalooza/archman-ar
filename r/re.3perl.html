re(3perl) دليل مبرمجي Perl re(3perl)

الاسم

re - Perl pragma لتغيير سلوك التعبيرات العادية

الملخص

use re 'taint';
($x) = ($^X =~ /^(.*)$/s);     # $x ملطخ هنا
$pat = '(?{ $foo = 1 })';
use re 'eval';
/foo${pat}bar/;                # لن يفشل (عندما لا يكون تحت مفتاح
                               # -T)
{
    no re 'taint';             # الافتراضي
    ($x) = ($^X =~ /^(.*)$/s); # $x ليس ملطخًا هنا
    no re 'eval';              # الافتراضي
    /foo${pat}bar/;            # غير مسموح به (مع أو بدون مفتاح -T)
}
use re 'strict';               # رفع تحذيرات لمزيد من الشروط
use re '/ix';
"FOO" =~ / foo /; # /ix ضمني
no re '/x';
"FOO" =~ /foo/; # /i ضمني فقط
use re 'debug';                # إخراج معلومات التصحيح أثناء
/^(.*)$/s;                     # وقت التجميع والتشغيل
use re 'debugcolor';           # نفس 'debug' ، ولكن مع إخراج ملون
...
use re qw(Debug All);          # نفس "use re 'debug'" ، ولكن يمكنك
                               # استخدام "Debug" مع أشياء أخرى
                               # غير 'All'
use re qw(Debug More);         # 'All' بالإضافة إلى إخراج المزيد من التفاصيل
no re qw(Debug ALL);           # تشغيل (تقريبًا) كل تصحيح re
                               # في هذا النطاق
use re qw(is_regexp regexp_pattern); # استيراد دوال مساعدة
my ($pat,$mods)=regexp_pattern(qr/foo/i);
if (is_regexp($obj)) {
    print "Got regexp: ",
        scalar regexp_pattern($obj); # بنفس طريقة perl في تحويله
}                                    # إلى سلسلة نصية ولكن دون عناء
                                     # re's الموصوفة.

(نستخدم $^X في هذه الأمثلة لأنها ملطخة افتراضيًا.)

الوصف

وضع 'taint'

عندما يكون "use re 'taint'" ساري المفعول ، وكان السلسلة النصية الملّطخة هدفًا للتعبير العادي ، فإن ذكريات التعبير العادي (أو القيم التي تم إرجاعها بواسطة عامل m// في سياق قائمة) تكون ملطخة. هذه الميزة مفيدة عندما لا تهدف عمليات التعبير العادي على البيانات الملطخة إلى استخراج سلاسل نصية فرعية آمنة ، بل إلى إجراء تحويلات أخرى.

وضع 'eval'

عندما يكون "use re 'eval'" ساري المفعول ، يُسمح للتعبير العادي بتضمين تأكيدات العرض الصفري "(?{ ... })" وتعبيرات فرعية مؤجلة "(??{ ... })" التي تم اشتقاقها من استيفاء المتغير ، بدلاً من ظهورها حرفيًا ضمن التعبير العادي. هذا غير مسموح به عادةً ، لأنه يمثل خطرًا أمنيًا محتملاً. لاحظ أن هذا الـ pragma يتم تجاهله عند الحصول على التعبير العادي من بيانات ملطخة ، أي أن التقييم غير مسموح به دائمًا مع التعبيرات العادية الملّطخة. انظر "(?{ code })" في perlre و "(??{ code })" في perlre.

لأغراض هذا الـ pragma ، لا يُعتبر استيفاء التعبيرات العادية المترجمة مسبقًا (أي نتيجة "qr//") استيفاءً للمتغير. وبالتالي:

/foo${pat}bar/

مسموح به إذا كان $pat تعبيرًا عاديًا مترجمًا مسبقًا ، حتى لو كان $pat يحتوي على تأكيدات "(?{ ... })" أو تعبيرات فرعية "(??{ ... })".

وضع 'strict'

لاحظ أن هذه ميزة تجريبية قد يتم تغييرها أو إزالتها في إصدار Perl مستقبلي.

عندما يكون "use re 'strict'" ساري المفعول ، يتم تطبيق فحوصات أكثر صرامة مما هو عليه الحال عادةً عند تجميع أنماط التعبير العادي. قد يؤدي هذا إلى رفع المزيد من التحذيرات أكثر من المعتاد ، وجعل المزيد من الأشياء قاتلة بدلاً من مجرد تحذيرات. الغرض من هذا هو اكتشاف الإبلاغ في وقت التجميع عن بعض الأشياء ، والتي قد تكون قانونية ، ولكن لديها احتمالية معقولة بأنها ليست النية الفعلية للمبرمج. هذا يشغل تلقائيًا فئة تحذيرات "regexp" (إذا لم تكن قيد التشغيل بالفعل) داخل نطاقها.

كمثال على شيء يتم التقاطه تحت 'strict' ، ولكن ليس بخلاف ذلك ، هو النمط

qr/\xABC/

يجب أن يتبع البناء "\x" بدون أقواس معقوفة رقمين ست عشريين بالضبط ؛ هذا يتبعه ثلاثة. يتم تقييم هذا حاليًا على أنه مكافئ لـ

qr/\x{AB}C/

أي الحرف الذي قيمة نقطة الترميز الخاصة به هي 0xAB ، متبوعًا بالحرف "C". ولكن نظرًا لأن "C" هو رقم ست عشري ، فهناك احتمال معقول أن تكون النية هي

qr/\x{ABC}/

أي الحرف الواحد عند 0xABC. تحت 'strict' ، يعد خطأ عدم اتباع "\x" برقمين ست عشريين بالضبط. عند عدم كونه تحت 'strict' ، يتم إنشاء تحذير إذا كان هناك رقم ست عشري واحد فقط ، ولا يتم رفع تحذير إذا كان هناك أكثر من اثنين.

من المتوقع أن يتطور ما يفعله 'strict' بالضبط بمرور الوقت مع اكتسابنا الخبرة معه. هذا يعني أن البرامج التي يتم تجميعها معه في Perl اليوم قد لا يتم تجميعها ، أو قد يكون لديها المزيد أو أقل من التحذيرات ، في إصدارات Perl المستقبلية. لا توجد وعود بتوافق رجعي فيما يتعلق به. أيضًا ، توجد بالفعل مقترحات لصيغة بديلة لتمكينه. لهذه الأسباب ، سيؤدي استخدامه إلى رفع تحذير من فئة "experimental::re_strict" ، ما لم يتم إيقاف تشغيل تلك الفئة.

لاحظ أنه إذا تمت إعادة تجميع نمط تم تجميعه داخل 'strict' ، على سبيل المثال عن طريق الاستيفاء في نمط آخر ، خارج 'strict' ، فلن يتم فحصه مرة أخرى للتأكد من صارمته. هذا لأنه إذا كان يعمل تحت الوضع الصارم ، فيجب أن يعمل تحت الوضع غير الصارم.

وضع '/flags'

عند تحديد "use re '/flags'a' ، 'aa' ، 'd' ، 'i' ، 'l' ، 'm' ، 'n' ، 'p' ، 's' ، 'u' ، 'x' ، و/أو 'xx'.

"no re '/flags"use re '/flagsعلى سبيل المثال ، إذا كنت تريد أن تكون جميع تعبيراتك العادية تحتوي على /msxx افتراضيًا ، فما عليك سوى وضع

use re '/msxx';

في بداية الكود الخاص بك.

تُلغي مجموعة الأحرف "/adul" flags بعضها البعض. لذا ، في هذا المثال ،

use re "/u";
"ss" =~ /\xdf/;
use re "/d";
"ss" =~ /\xdf/;

يقوم "use re" الثاني ضمنيًا بـ "no re '/u'".

بالمثل ،

use re "/xx";   # مضاعف-x
...
use re "/x";    # x واحد من هنا فصاعدًا
...

يؤدي تشغيل إحدى علامات مجموعة الأحرف باستخدام "use re" إلى إعطاء الأسبقية لـ "locale" pragma و "feature" 'unicode_strings' ، للتعبيرات العادية. يؤدي إيقاف تشغيل إحدى هذه العلامات عند تنشيطها إلى العودة إلى السلوك المحدد بواسطة أي pragmata أخرى قيد النطاق. على سبيل المثال:

use feature "unicode_strings";
no re "/u"; # لا يفعل شيئا
use re "/l";
no re "/l"; # يعود إلى سلوك unicode_strings

يتم تطبيق العلامات الافتراضية على المكان الذي يتم فيه تجميع النمط باستخدام الاستثناء من علامة "/x" ، والتي لا يتم تطبيقها على الأنماط المجمعة من الوسائط النصية لـ "split". وبالتالي فإن `use re "/x";` لا يؤثر على سلوك "split " "" ولكنه يؤثر على سلوك "split / /".

وضع 'debug'

عندما يكون "use re 'debug'" ساري المفعول ، يقوم perl بإصدار رسائل تصحيح عند تجميع واستخدام التعبيرات العادية. يكون الإخراج هو نفسه الذي تم الحصول عليه عن طريق تشغيل "-DDEBUGGING"-مُمكَّن

مفسر perl باستخدام الخيار -Dr. قد يكون هذا الخيار ضخمًا جدًا اعتمادًا على تعقيد المطابقة. استخدام "debugcolor" بدلاً من "debug" يُمكّن شكلاً من أشكال الإخراج الذي يمكن استخدامه للحصول على عرض ملون على الأطراف التي تفهم تسلسلات ألوان termcap. قم بتعيين $ENV{PERL_RE_TC} إلى قائمة مفصولة بفواصل من خصائص "termcap" لاستخدامها لتسليط الضوء على السلاسل النصية تشغيل/إيقاف، والجزء قبل النقطة تشغيل/إيقاف. انظر "تصحيح الأخطاء في التعبيرات العادية" في perldebug لمزيد من المعلومات.

لاحظ أن التنسيق الدقيق لوضع "debug" ليس يعتبر واجهة برمجة تطبيقات مدعومة رسميًا في Perl. إنه مخصص لتصحيح الأخطاء فقط وقد يتغير كما ترى فريق التطوير الأساسي مناسبًا دون إشعار أو إيقاف في أي إصدار من Perl، رئيسي أو ثانوي. أي توثيق للإخراج هو استشاري فقط.

اعتبارًا من 5.9.5، يكون التوجيه "use re 'debug'" وما يعادله له نطاق معجمي، كما هو الحال مع التوجيهات الأخرى. ومع ذلك، فإن لهما تأثيرات على وقت التجميع ووقت التشغيل.

انظر "الوحدات العملية" في perlmodlib.

'Debug' mode

بالمثل، ينتج "use re 'Debug'" إخراج تصحيح الأخطاء، والفرق هو أنه يسمح بضبط دقيق للإخراج الذي سيتم إصداره. تنقسم الخيارات إلى ثلاث مجموعات، تلك المتعلقة بالتجميع، وتلك المتعلقة بالتنفيذ، وتلك المتعلقة بأغراض خاصة.

لاحظ أن الخيارات المقدمة ضمن وضع "Debug" والتنسيق الدقيق للإخراج الذي تنشئه ليس معتبرًا واجهة برمجة تطبيقات مدعومة رسميًا في Perl. إنه مخصص لتصحيح الأخطاء فقط وقد يتغير كما ترى فريق التطوير الأساسي مناسبًا دون إشعار أو إيقاف في أي إصدار من Perl، رئيسي أو ثانوي. أي توثيق للتنسيق أو الخيارات المتاحة هو استشاري فقط ويخضع للتغيير دون إشعار.

الخيارات هي كما يلي:

خيارات متعلقة بالتجميع
COMPILE
يقوم بتشغيل جميع خيارات تصحيح الأخطاء غير الإضافية المتعلقة بالتجميع.
PARSE
يقوم بتشغيل إخراج تصحيح الأخطاء المتعلق بعملية تحليل النمط.
OPTIMISE
يمكّن الإخراج المتعلق بمرحلة تحسين التجميع.
TRIEC
معلومات مفصلة حول تجميع الشجرة الثلاثية (trie compilation).
DUMP
يفرغ البرنامج النهائي بعد تجميعه وتحسينه.
FLAGS
يفرغ العلامات المرتبطة بالبرنامج.
TEST
يطبع الإخراج المخصص لاختبار الأجزاء الداخلية لعملية التجميع.
خيارات متعلقة بالتنفيذ
EXECUTE
يقوم بتشغيل جميع خيارات تصحيح الأخطاء غير الإضافية المتعلقة بالتنفيذ.
MATCH
يقوم بتشغيل تصحيح الأخطاء لحلقة المطابقة الرئيسية.
TRIEE
تصحيح إضافي للأخطاء حول كيفية تنفيذ الأشجار الثلاثية.
INTUIT
يمكّن تصحيح الأخطاء لتحسينات نقطة البداية.
خيارات تصحيح أخطاء إضافية
EXTRA
يقوم بتشغيل جميع خيارات تصحيح الأخطاء "الإضافية".
BUFFERS
يمكّن تصحيح الأخطاء لتخزين مجموعات الالتقاط أثناء المطابقة. تحذير، يمكن أن ينتج هذا عن إخراج كبير للغاية.
TRIEM
يمكّن تصحيح الأخطاء المحسن للأشجار الثلاثية. يعزز كل من TRIEE و TRIEC.
STATE
يمكّن تصحيح الأخطاء للحالات في المحرك.
STACK
يمكّن تصحيح الأخطاء لمكدس الاستدعاءات العودية في المحرك. تمكين أو تعطيل هذا الخيار يقوم تلقائيًا بنفس الشيء لتصحيح الأخطاء للحالات أيضًا. يمكن أن يكون الإخراج من هذا كبيرًا جدًا.
GPOS
يمكّن تصحيح الأخطاء للمعدل \G.
OPTIMISEM
يمكّن تصحيح الأخطاء المحسن للتحسين وتحسينات نقطة البداية. ربما لا يكون مفيدًا إلا عند تصحيح أخطاء محرك التعبيرات العادية نفسه.
DUMP_PRE_OPTIMIZE
يمكّن تفريغ النمط المجمع قبل مرحلة التحسين.
WILDCARD
عندما يصادف Perl نمطًا فرعيًا عشوائيًا (انظر "الرموز العشوائية في قيم الخصائص" في perlunicode)، فإنه يعلق تجميع النمط الرئيسي، ويجمع النمط الفرعي، ثم يطابق ذلك مع جميع الاحتمالات القانونية لتحديد نقاط التعليمات البرمجية الفعلية التي يطابقها النمط الفرعي. بعد ذلك، يضيف هذه إلى النمط الرئيسي، ويستمر في تجميعه.

قد ترغب بشدة في رؤية كيفية تجميع نمطك الفرعي، ولكن من غير المحتمل أن يكون من المفيد لك رؤية كيفية مطابقة Perl لذلك مع جميع الاحتمالات القانونية، حيث أن ذلك تحت سيطرة Perl، وليس أنت. لذلك، فإن معلومات تصحيح الأخطاء لجزء التجميع هي كما هو محدد بواسطة الخيارات الأخرى، ولكن يتم قمع إخراج تصحيح الأخطاء لجزء المطابقة عادةً.

يمكنك استخدام خيار WILDCARD لتمكين إخراج تصحيح الأخطاء لهذه المطابقة الفرعية للنمط. حذر! يمكن أن يؤدي هذا إلى مخرجات ضخمة، وقد لا يكون له معنى كبير بالنسبة لك ما يفعله Perl ولماذا. ولكنه قد يكون مفيدًا لك لمعرفة سبب عدم سير الأمور بالطريقة التي تتوقعها.

لاحظ أن هذا الخيار وحده لا يتسبب في أي إخراج لمعلومات تصحيح الأخطاء. ما يفعله هو إيقاف القمع الطبيعي لمعلومات تصحيح الأخطاء المتعلقة بالتنفيذ أثناء جزء المطابقة من تجميع الرموز العشوائية. يجب عليك أيضًا تحديد معلومات تصحيح الأخطاء المتعلقة بالتنفيذ التي تريدها، مثل تضمين خيار EXECUTE أيضًا.

هذه اختصارات مفيدة لتوفير الكتابة.
ALL
يمكّن جميع الخيارات مرة واحدة باستثناء BUFFERS و WILDCARD و DUMP_PRE_OPTIMIZE. (للحصول على كل خيار على حدة دون استثناء، استخدم كل من ALL و EXTRA، أو بدءًا من 5.30 على مفسر perl ممكّن ، استخدم مفاتيح سطر الأوامر -Drv.)
All
يمكّن DUMP وجميع خيارات التنفيذ غير الإضافية. مكافئ لـ: 
use re 'debug';
MORE
More
يمكّن الخيارات التي تم تمكينها بواسطة "All"، بالإضافة إلى STATE و TRIEC و TRIEM.

وظائف قابلة للتصدير

اعتبارًا من perl 5.9.5، تحتوي الوحدة "re" على عدد من وظائف الأدوات المساعدة التي يمكن تصديرها اختياريًا إلى مساحة أسماء المتصل. وهي مدرجة أدناه.

is_regexp($ref)
يعيد صحيحًا إذا كان الوسيط عبارة عن تعبير عادي مجمع تم إرجاعه بواسطة "qr//"، ويعيد خطأ إذا لم يكن كذلك.

لن يتم خداع هذه الوظيفة بواسطة التحميل الزائد أو البركة (blessing). من الناحية الداخلية، تستخرج هذه الوظيفة مؤشر التعبير العادي من بنية PERL_MAGIC_qr حتى لا يمكن خداعها.

regexp_pattern($ref)
إذا كان الوسيط عبارة عن تعبير عادي مجمع تم إرجاعه بواسطة "qr//"، فإن هذه الوظيفة تعيد النمط.

في سياق قائمة، تعيد قائمة ذات عنصرين، العنصر الأول يحتوي على النمط والعنصر الثاني يحتوي على المعدلات المستخدمة عند تجميع النمط.

my ($pat, $mods) = regexp_pattern($ref);

في سياق قياسي، تعيد نفس ما سيعيده Perl عند تحويل تعبير "qr//" خام يحتوي على نفس النمط بداخله إلى سلسلة نصية. إذا لم يكن الوسيط مرجعًا مجمعًا، فإن هذه الروتينية تعيد خطأ ولكن معرفًا في سياق قياسي، والقائمة الفارغة في سياق قائمة. وبالتالي فإن التالي

if (regexp_pattern($ref) eq '(?^i:foo)')

سيكون خاليًا من التحذيرات بغض النظر عن قيمة $ref الفعلية.

مثل "is_regexp"، لن يتم خداع هذه الوظيفة بواسطة التحميل الزائد أو البركة للكائن.

regname($name,$all)
يعيد محتويات مخزن مؤقت مسمى للمطابقة الناجحة الأخيرة. إذا كان $all صحيحًا، فإنه يعيد مرجع مصفوفة يحتوي على مدخل واحد لكل مخزن مؤقت، وإلا فإنه يعيد المخزن المؤقت المحدد الأول.
regnames($all)
يعيد قائمة بجميع المخازن المؤقتة المسماة المحددة في المطابقة الناجحة الأخيرة. إذا كان $all صحيحًا، فإنه يعيد جميع الأسماء المحددة، وإذا لم يكن كذلك، فإنه يعيد الأسماء التي شاركت في المطابقة فقط.

perl -- perl regular expression compiler

NAME

re -- Perl regular expression compiler

SYNOPSIS

    use re qw/ ... /;

DESCRIPTION

This module provides access to the internals of Perl's regular expression engine. It is intended for use by module authors who need to understand how Perl compiles and optimizes regular expressions.

def_execaddr($ref)
Used by the compiler to set up the Perl code that will be executed in the event of a match.
def_execrun($ref)
Used by the compiler to set up the Perl code that will be executed when a regular expression is run.
use($module:?)
This is the function that the regular expression compiler calls to load modules that are specified with the "use" pragma within a regular expression. For example, in "qr/ use My::Module; ... /x", the compiler will call re::use("My::Module").
local_rank($ref)
This returns the optimisation rank of the regular expression. A higher rank means that the regular expression is more likely to be useful for optimisation. For example, a pattern like "/^$/" has a high rank, while "/a|b|c|d|e|f|g|h|i|j|k|l/" has a low rank. This is primarily used by the compiler itself to decide which parts of a regular expression can be optimised away.
tie_regexp($regexp: $str:?)
This function is used to tie a regular expression to a string. It's used internally by Perl, and is not generally useful outside of that context. It is used to implement the behaviour of "tied()" on regular expression matches.
eval_sv($sv:?)
This is used to evaluate a scalar value within the context of a regular expression match. It's used internally by the compiler when it encounters constructs like "(?{ ... })" or "(??{ ... })".
overlapped($ref)
If the argument is a compiled regular expression as returned by "qr//", then this function returns a boolean value indicating whether the regular expression engine should look for overlapped matches. This is typically only true for the "...(*ACCEPT*) ..." and "...(*FAIL*) ..." features.
submatch_context($ref)
Returns a string indicating the context in which submatches are being evaluated. Possible values include "MATCH", "SEARCH", and "NONE".
sv_export_pairs($ref)
If the argument is a compiled regular expression as returned by "qr//", then this function returns a reference to an array of pairs. Each pair contains a name and a value that is exported by the regular expression's private namespace. This is used when quantifiers like "*?" are used in a non-greedy fashion.
eval_rules($ref)
If the argument is a compiled regular expression as returned by "qr//", then this function returns a hashref containing the evaluation rules for the regular expression. These rules dictate how the regular expression is evaluated at runtime.
no_unicode_check($ref)
If the argument is a compiled regular expression as returned by "qr//", then this function returns a boolean value indicating whether unicode checks should be disabled for this regular expression. This is used for backwards compatibility with older versions of Perl.
set_re_strict($value)
Sets the strictness level for regular expression compilation. A higher value means stricter checking. This is mostly used internally by the compiler.
set_re_strict_in_code($value:?)
Similar to "set_re_strict", but this function affects the strictness level only within the code block of a regular expression. This allows for finer control over strictness.
parse_qr($qr_source:?)
This function takes the source code of a regular expression (as a string) and parses it, returning a compiled regular expression object. This is useful for dynamically creating and compiling regular expressions.
qr_data($ref)
If the argument is a compiled regular expression as returned by "qr//", then this function returns a hashref containing internal data structures used by the regular expression compiler. This is primarily for debugging and analysis purposes.
set_compile_options($ref:?)
This function allows you to set various compile-time options for regular expressions. The argument is a hashref containing the options to set.
get_compile_options($ref:?)
This function retrieves the current compile-time options for regular expressions. If a compiled regular expression object is passed as an argument, it returns the options specific to that expression.
check_match($regex: $string:?)
This function attempts to perform a regular expression match against a given string. It returns true if the match succeeds, false otherwise. It's useful for testing the behavior of regular expressions without actually capturing the results.
check_string($regex: $string:?)
Similar to "check_match", but this function specifically checks if the entire string matches the regular expression. It returns true if the entire string matches, false otherwise.
set_parameter($name: $value:?)
Allows setting internal parameters of the regex engine. The available parameters and their effects are implementation-dependent and may change between Perl versions. Use with extreme caution.
regmust($ref)
If the argument is a compiled regular expression as returned by "qr//", then this function returns what the optimiser considers to be the longest anchored fixed string and longest floating fixed string in the pattern.

A fixed string is defined as being a substring that must appear for the pattern to match. An anchored fixed string is a fixed string that must appear at a particular offset from the beginning of the match. A floating fixed string is defined as a fixed string that can appear at any point in a range of positions relative to the start of the match. For example,

my $qr = qr/here .* there/x;
my ($anchored, $floating) = regmust($qr);
print "anchored:'$anchored'\nfloating:'$floating'\n";

results in

anchored:'here'
floating:'there'

Because the "here" is before the ".*" in the pattern, its position can be determined exactly. That's not true, however, for the "there"; it could appear at any point after where the anchored string appeared. Perl uses both for its optimisations, preferring the longer, or, if they are equal, the floating.

NOTE: This may not necessarily be the definitive longest anchored and floating string. This will be what the optimiser of the Perl that you are using thinks is the longest. If you believe that the result is wrong please report it via the perlbug utility.

optimization($ref)
If the argument is a compiled regular expression as returned by "qr//", then this function returns a hashref of the optimization information discovered at compile time, so we can write tests around it. If any other argument is given, returns "undef".

The hash contents are expected to change from time to time as we develop new ways to optimize - no assumption of stability should be made, not even between minor versions of perl.

For the current version, the hash will have the following contents:

minlen
An integer, the least number of characters in any string that can match.
minlenret
An integer, the least number of characters that can be in $& after a match. (Consider eg " /ns(?=\d)/ ".)
gofs
An integer, the number of characters before pos() to start match at.
noscan
A boolean, "TRUE" to indicate that any anchored/floating substrings found should not be used. (CHECKME: apparently this is set for an anchored pattern with no floating substring, but never used.)
isall
A boolean, "TRUE" to indicate that the optimizer information is all that the regular expression contains, and thus one does not need to enter the regexp runtime engine at all.
anchor SBOL
A boolean, "TRUE" if the pattern is anchored to start of string.
anchor MBOL
A boolean, "TRUE" if the pattern is anchored to any start of line within the string.
anchor GPOS
A boolean, "TRUE" if the pattern is anchored to the end of the previous match.
skip
A boolean, "TRUE" if the start class can match only the first of a run.
implicit
A boolean, "TRUE" if a "/.*/" has been turned implicitly into a "/^.*/".
anchored/floating
A byte string representing an anchored or floating substring respectively that any match must contain, or undef if no such substring was found, or if the substring would require utf8 to represent.
anchored utf8/floating utf8
A utf8 string representing an anchored or floating substring respectively that any match must contain, or undef if no such substring was found, or if the substring contains only 7-bit ASCII characters.
anchored min offset/floating min offset
An integer, the first offset in characters from a match location at which we should look for the corresponding substring.
anchored max offset/floating max offset
An integer, the last offset in characters from a match location at which we should look for the corresponding substring.

Ignored for anchored, so may be 0 or same as min.
