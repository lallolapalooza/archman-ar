الاسم

rtcInitIntersectContext - تهيئة سياق التقاطع

الملخص

#include <embree3/rtcore.h>
enum RTCIntersectContextFlags
{
  RTC_INTERSECT_CONTEXT_FLAG_NONE,
  RTC_INTERSECT_CONTEXT_FLAG_INCOHERENT,
  RTC_INTERSECT_CONTEXT_FLAG_COHERENT,
};
struct RTCIntersectContext
{
  enum RTCIntersectContextFlags flags;
  RTCFilterFunctionN filter;
  #if RTC_MAX_INSTANCE_LEVEL_COUNT > 1
    unsigned int instStackSize;
  #endif
  unsigned int instID[RTC_MAX_INSTANCE_LEVEL_COUNT];
  #if RTC_MIN_WIDTH
    float minWidthDistanceFactor;
  #endif
};
void rtcInitIntersectContext(
  struct RTCIntersectContext* context
);

الوصف

يتم دعم سياق تقاطع لكل شعاع استعلام (من نوع RTCIntersectContext) يمكن استخدامه لتكوين علامات التقاطع (عضو flags)، وتحديد دالة استدعاء مرشح (عضو filter)، وتحديد سلسلة معرفات المثيل الحالي (أعضاء instID و instStackSize)، وإرفاق بيانات تعسفية بالاستعلام (مثل بيانات لكل شعاع).

تقوم دالة rtcInitIntersectContext بتهيئة السياق إلى قيم افتراضية ويجب استدعاؤها لتهيئة كل سياق تقاطع. يتم تضمين هذه الدالة، مما يقلل من الحمل الزائد ويسمح بتحسينات المترجم.

يمكن استخدام علامة سياق التقاطع لضبط سلوك خوارزمية الاجتياز. استخدام علامات RTC_INTERSECT_CONTEXT_FLAG_INCOHERENT يستخدم خوارزمية اجتياز محسّنة للأشعة غير المترابطة (افتراضيًا)، بينما تستخدم RTC_INTERSECT_CONTEXT_FLAG_COHERENT خوارزمية اجتياز محسّنة للأشعة المترابطة (مثل أشعة الكاميرا الأولية).

يمكن الحصول على أفضل أداء للأشعة الأولية باستخدام واجهة برمجة تطبيقات تدفق الأشعة وتعيين علامة سياق التقاطع إلى RTC_INTERSECT_CONTEXT_FLAG_COHERENT. بالنسبة للأشعة الثانوية، من الأفضل عادةً استخدام علامة RTC_INTERSECT_CONTEXT_FLAG_INCOHERENT، إلا إذا كانت الأشعة معروفة بأنها مترابطة جدًا أيضًا (مثل أشعة الشفافية الأولية).

يمكن تحديد دالة مرشح داخل السياق. يتم استدعاء دالة المرشح هذه كمرحلة تصفية ثانية بعد استدعاء دالة التقاطع أو الإغلاق لكل هندسة. الأشعة التي اجتازت مرحلة التصفية الأولى فقط تكون صالحة في مرحلة التصفية الثانية هذه. يمكن أن يكون وجود دالة مرشح لكل استعلام شعاع مفيدًا لتنفيذ تعديلات على سلوك الاستعلام، مثل تجميع كل الضربات أو تجميع الشفافيات. يجب تمكين دعم دالة مرشح السياق لمشهد باستخدام علامة المشهد RTC_SCENE_FLAG_CONTEXT_FILTER_FUNCTION. في حالة المثيلات، يجب تمكين هذه الميزة أيضًا لكل مشهد تم إنشاؤه كمثيل.

يتحكم عامل minWidthDistanceFactor في الحجم المستهدف لنصف أقطار المنحنى عند تمكين ميزة الحد الأدنى للعرض. يرجى الاطلاع على دالة [rtcSetGeometryMaxRadiusScale] لمزيد من التفاصيل حول ميزة الحد الأدنى للعرض.

من المضمون أن يتم تمرير المؤشر إلى سياق التقاطع المقدم إلى استعلام شعاع مباشرة إلى دوال الاستدعاء المسجلة. بهذه الطريقة، من الممكن إرفاق بيانات تعسفية بنهاية سياق التقاطع، مثل حمولة لكل شعاع.

يرجى ملاحظة أنه لا يُضمن تمرير مؤشر الشعاع إلى دوال الاستدعاء، وبالتالي لا يمكن قراءة بيانات إضافية من مؤشر الشعاع المقدم إلى دوال الاستدعاء.
