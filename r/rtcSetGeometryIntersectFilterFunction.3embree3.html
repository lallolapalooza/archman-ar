الاسم

rtcSetGeometryIntersectFilterFunction - يحدد دالة فلترة التقاطع
  للشكل الهندسي

الملخص

#include <embree3/rtcore.h>
struct RTCFilterFunctionNArguments
{
  int* valid;
  void* geometryUserPtr;
  const struct RTCIntersectContext* context;
  struct RTCRayN* ray;
  struct RTCHitN* hit;
  unsigned int N;
};
typedef void (*RTCFilterFunctionN)(
  const struct RTCFilterFunctionNArguments* args
);
void rtcSetGeometryIntersectFilterFunction(
  RTCGeometry geometry,
  RTCFilterFunctionN filter
);

الوصف

تقوم الدالة rtcSetGeometryIntersectFilterFunction بتسجيل دالة استدعاء ل فلترة التقاطع (وسيطة الفلتر) للشكل الهندسي المحدد (وسيطة الشكل الهندسي).

يمكن تسجيل دالة استدعاء واحدة فقط لكل شكل هندسي، وتؤدي الاستدعاءات اللاحقة إلى الكتابة فوق دالة الاستدعاء المعينة سابقًا. يؤدي تمرير NULL ك مؤشر للدالة إلى تعطيل دالة الاستدعاء المسجلة.

يتم استدعاء دالة فلترة التقاطع المسجلة لكل اصطدام يتم مواجهته أثناء استعلامات الأشعة من نوع rtcIntersect ويمكنها قبول هذا الاصطدام أو رفضه. يمكن استخدام هذه الميزة لتحديد صورة ظلية لبدائي ورفض الاصطدامات التي تقع خارج الصورة الظلية. على سبيل المثال، يمكن نمذجة ورقة شجرة باستخدام نسيج ألفا يقرر ما إذا كانت نقاط الاصطدام تقع داخل أو خارج الورقة.

إذا تم تعيين وضع RTC_BUILD_QUALITY_HIGH، فقد يتم استدعاء دوال الفلترة عدة مرات لنفس الاصطدام البدائي. علاوة على ذلك، قد تقوم الأشعة التي تصطدم تمامًا بالحافة بالإبلاغ عن اصطدامين لنفس السطح. بالنسبة لحالات الاستخدام المحددة، قد يحتاج التطبيق إلى التحايل على هذا القيد عن طريق تجميع أزواج (geomID/primID) التي تم الإبلاغ عنها بالفعل وتجاهل التكرارات.

تتلقى دالة الاستدعاء فلتر من نوع RTCFilterFunctionN عددًا من الوسائط من خلال بنية RTCFilterFunctionNArguments. يشير المؤشر valid في هذه البنية إلى قناع صلاحية صحيح (0 يعني غير صالح و -1 يعني صالح). العضو geometryUserPtr هو مؤشر مستخدم تم تعيينه اختياريًا لكل شكل هندسي من خلال دالة rtcSetGeometryUserData. يشير العضو context إلى سياق التقاطع الذي تم تمريره إلى دالة استعلام الأشعة. يشير المؤشر ray إلى N أشعة بتنسيق SOA. يشير المؤشر hit إلى N اصطدامات بتنسيق SOA لاختبارها. المعلمة N هي عدد الأشعة والاصطدامات في ray و hit. يتم توفير مسافة الاصطدام كقيمة tfar للشعاع. إذا كان الشكل الهندسي المصطدم مجسمًا، فإن العضو instID للشعاع يكون صالحًا، والشعاع والاصطدام المحتمل يكونان في مساحة الكائن.

تتمثل مهمة دالة استدعاء الفلتر في التحقق من كل شعاع صالح مما إذا كان يريد قبول أو رفض الاصطدام المقابل. لرفض الاصطدام، ما على دالة استدعاء الفلتر سوى كتابة 0 إلى قناع الصلاحية الصحيح للشعاع المقابل. لقبول الاصطدام، ما عليها سوى ترك قناع الصلاحية مضبوطًا على -1. يُسمح لدالة الفلتر أيضًا بتغيير الاصطدام وتقليل قيمة tfar للشعاع، ولكن يجب ألا تعدل بيانات الشعاع الأخرى ولا أي مكونات غير نشطة للشعاع أو الاصطدام.

يمكن لتنفيذ دالة الفلتر اختيار تنفيذ مسار رمز واحد يستخدم دوال مساعدة للوصول إلى الأشعة RTCRay_XXX ودوال مساعدة للوصول إلى الاصطدامات RTCHit_XXX للوصول إلى بيانات الشعاع والاصطدام. بدلاً من ذلك، يمكن للرمز الانتقال إلى تطبيقات محسّنة لأحجام محددة من N ونسخ مدخلات الشعاع والاصطدام إلى أنواع الحزم المناسبة.
