الاسم

rtcSetGeometryPointQueryFunction - تعيين دالة استدعاء الاستعلام النقطي لهندسة

الملخص

#include <embree4/rtcore.h>
struct RTCPointQueryFunctionArguments
{
  // كائن الاستعلام (فضاء عالمي) الذي تم تمريره كوسيط لـ rtcPointQuery.
  struct RTCPointQuery* query;
  // يستخدم لبيانات الإدخال/الإخراج للمستخدم. لن يتم قراءته أو تعديله داخليًا.
  void* userPtr;
  // معرف البدائي ومعرف الهندسة للبدائي
  unsigned int  primID;        
  unsigned int  geomID;    
  // سياق مكدس التحويل ومعرف المثيل
  struct RTCPointQueryContext* context;
  // عامل القياس الذي يشير إلى ما إذا كان تحويل المثيل الحالي
  // هو تحويل تشابه.
  float similarityScale;
};
typedef bool (*RTCPointQueryFunction)(
  struct RTCPointQueryFunctionArguments* args
);
void rtcSetGeometryPointQueryFunction(
  RTCGeometry geometry,
  RTCPointQueryFunction queryFunc
);

الوصف

تقوم الدالة rtcSetGeometryPointQueryFunction بتسجيل دالة استدعاء الاستعلام النقطي (الوسيط queryFunc) للهندسة المحددة (الوسيط geometry).

يمكن تسجيل دالة استدعاء واحدة فقط لكل هندسة، وستؤدي الاستدعاءات اللاحقة إلى الكتابة فوق دالة الاستدعاء التي تم تعيينها سابقًا. تمرير NULL كـ مؤشر للدالة يعطل دالة الاستدعاء المسجلة.

يتم استدعاء دالة الاستدعاء المسجلة بواسطة [rtcPointQuery] لكل بدائي من الهندسة يتقاطع مع مجال الاستعلام النقطي المقابل. تحصل دالة الاستدعاء من النوع RTCPointQueryFunction على عدد من الوسائط من خلال الهيكل RTCPointQueryFunctionArguments. كائن الاستعلام هو كائن الاستعلام النقطي الأصلي الذي تم تمريره إلى [rtcPointQuery]، usrPtr هو مؤشر عشوائي لتمرير البيانات وإدخالها وتخزين نتائج دالة الاستدعاء. يمكن استخدام primID و geomID والسياق (انظر [rtcInitPointQueryContext] للحصول على التفاصيل) لتحديد بيانات هندسة البدائي.

يمكن أيضًا تمرير RTCPointQueryFunction مباشرة كوسيط لـ [rtcPointQuery]. في هذه الحالة، يتم استدعاء الاستدعاء لجميع البدائيات في المشهد التي تتقاطع مع مجال الاستعلام. إذا تم تمرير دالة استدعاء كوسيط لـ [rtcPointQuery] وتم تعيين دالة استدعاء (ربما مختلفة) لهندسة باستخدام [rtcSetGeometryPointQueryFunction]، فسيتم استدعاء كلتا الدالتين، وسيتم استدعاء دالة الاستدعاء التي تم تمريرها إلى [rtcPointQuery] قبل دالة الاستدعاء الخاصة بالهندسة.

إذا تم استخدام المثيلات، يشير المعلمة simliarityScale إلى ما إذا كان تحويل المثيل الحالي (العنصر العلوي في المكدس في السياق) هو تحويل تشابه أم لا. تحويلات التشابه تتكون من الترجمة والدوران والقياس الموحد، وإذا كانت مصفوفة M تحدد تحويل تشابه، فهناك عامل قياس D بحيث يكون لكل x,y: dist(Mx, My) = D * dist(x, y). في هذه الحالة، يكون المعلمة scalingFactor هو عامل القياس D، وإلا فهو 0. مقياس تشابه صالح (similarityScale > 0) يسمح بحساب معلومات المسافة في فضاء المثيل وقياس المسافات إلى الفضاء العالمي (على سبيل المثال، لتحديث نصف قطر الاستعلام، انظر أدناه) عن طريق قسمة المسافة في فضاء المثيل على مقياس التشابه. إذا لم يكن تحويل المثيل الحالي تحويل تشابه (similarityScale هو 0)، فيجب إجراء حساب المسافة في الفضاء العالمي لضمان الدقة. في هذه الحالة، يجب استخدام التحويلات من المثيل إلى العالم المقدمة مع السياق لتحويل بيانات البدائي إلى الفضاء العالمي. بخلاف ذلك، يمكن تحويل موقع الاستعلام إلى فضاء المثيل، وهو ما قد يكون أكثر كفاءة. إذا لم يكن هناك تحويل مثيل، يكون مقياس التشابه 1.

سيتم استدعاء دالة الاستدعاء المحتملة للبدائيات خارج مجال الاستعلام لسببين: أولاً، يتم استدعاء الاستدعاء لجميع البدائيات داخل عقدة ورقة BVH نظرًا لعدم تحديد أي بيانات هندسة للبدائيات داخليًا، وبالتالي لا يتم استبعاد البدائيات الفردية (فقط مربعاتها المحيطة المجمعة). ثانيًا، في حالة استخدام تحويلات غير تشابه، يتم تقريب مجال الاستعلام الإهليلجي الناتج (في فضاء المثيل) بواسطة مربع حدوده المحاذي للمحاور داخليًا، وبالتالي قد تتقاطع العقد الداخلية التي لا تتقاطع مع المجال الأصلي مع المربع المحيط التقريبي، مما يؤدي إلى استدعاءات غير ضرورية. في أي حال، الاستدعاءات متحفظة، أي أنه إذا كان البدائي داخل مجال الاستعلام، فسيتم استدعاء استدعاء، ولكن العكس ليس ضروريًا.

لتحقيق الكفاءة، يمكن تقليل نصف قطر كائن الاستعلام (في الفضاء العالمي) داخل دالة الاستدعاء لتحسين استبعاد الهندسة أثناء اجتياز BVH. إذا تم تحديث نصف قطر الاستعلام، فيجب أن تعيد دالة الاستدعاء true لإصدار تحديث لمعلومات الاجتياز الداخلية. زيادة نصف القطر أو تعديل الوقت أو موقع الاستعلام يؤدي إلى سلوك غير محدد.

داخل دالة الاستدعاء، من الآمن استدعاء [rtcPointQuery] مرة أخرى، على سبيل المثال عند تنفيذ المثيلات يدويًا. في هذه الحالة، يجب وضع تحويل المثيل على المكدس في السياق. ستقوم Embree داخليًا بحساب معلومات الاستعلام النقطي في فضاء المثيل باستخدام العنصر العلوي في المكدس في السياق عند استدعاء [rtcPointQuery].

للتنفيذ المرجعي لاجتياز أقرب نقطة لشبكات المثلثات باستخدام المثيلات والمثيلات المعرفة من قبل المستخدم، انظر البرنامج التعليمي [ClosestPoint].
