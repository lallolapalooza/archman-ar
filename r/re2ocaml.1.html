الاسم

re2ocaml - إنشاء محللات معجمية سريعة لـ OCaml

الموجز

re2ocaml [ خيارات ] [ تحذيرات ] إدخال

يمكن أن يكون الإدخال إما ملفًا أو - للإدخال القياسي.

مقدمة

يعمل re2ocaml كمعالج مسبق. يقرأ ملف الإدخال (وهو عادةً برنامج بلغة OCaml، ولكنه يمكن أن يكون أي شيء) ويبحث عن كتل من التعليمات البرمجية محاطة بعلامات بداية/نهاية خاصة. يتم نسخ النص خارج هذه الكتل حرفيًا إلى ملف الإخراج. تتم معالجة محتويات الكتل بواسطة re2ocaml. يقوم بترجمتها إلى تعليمات برمجية بلغة OCaml ويخرج التعليمات البرمجية المولدة في مكان الكتلة.

إليك مثال لبرنامج صغير يتحقق مما إذا كان سلسلة معينة تحتوي على رقم عشري:

(* re2ocaml $INPUT -o $OUTPUT -i *)
open String
type state = {
    yyinput: string;
    mutable yycursor: int;
}
%{
    re2c:YYFN = ["lex;bool", "yyrecord;state"];
    re2c:yyfill:enable = 0;
    [1-9][0-9]* { true }
    *           { false }
%}
let main () =
    let st = {yyinput = "1234"; yycursor = 0}
    in if not (lex st) then raise (Failure "error")
let _ = main ()

في الإخراج، استبدل re2ocaml الكتلة في المنتصف بالتعليمات البرمجية المولدة:

(* تم الإنشاء بواسطة re2ocaml *)
(* re2ocaml $INPUT -o $OUTPUT -i *)
open String
type state = {
    yyinput: string;
    mutable yycursor: int;
}
let rec yy0 (yyrecord : state) : bool =
    let yych = unsafe_get yyrecord.yyinput yyrecord.yycursor in
    yyrecord.yycursor <- yyrecord.yycursor + 1;
    match yych with
        | '1'..'9' -> (yy2 [@tailcall]) yyrecord
        | _ -> (yy1 [@tailcall]) yyrecord
and yy1 (yyrecord : state) : bool =
    false
and yy2 (yyrecord : state) : bool =
    let yych = unsafe_get yyrecord.yyinput yyrecord.yycursor in
    match yych with
        | '0'..'9' ->
            yyrecord.yycursor <- yyrecord.yycursor + 1;
            (yy2 [@tailcall]) yyrecord
        | _ -> (yy3 [@tailcall]) yyrecord
and yy3 (yyrecord : state) : bool =
    true
and lex (yyrecord : state) : bool =
    (yy0 [@tailcall]) yyrecord
let main () =
    let st = {yyinput = "1234"; yycursor = 0}
    in if not (lex st) then raise (Failure "error")
let _ = main ()

أساسيات

يتكون برنامج re2ocaml من تسلسل من الكتل تتخللها تعليمات برمجية باللغة المستهدفة. قد تحتوي الكتلة على تعريفات، تكوينات، قواعد، إجراءات وتوجيهات بأي ترتيب:

الاسم = تعبير-عادي ;
تعريف يربط الاسم بـ التعبير-العادي. يمكن للأسماء أن تحتوي على أحرف أبجدية رقمية وشرطة سفلية. يعطي قسم التعبيرات العادية نظرة عامة على صيغة re2ocaml للتعبيرات العادية. بمجرد تعريفه، يمكن استخدام الاسم في تعبيرات عادية أخرى وفي القواعد. التكرار في التعريفات المسماة غير مسموح به، ويجب تعريف كل اسم قبل استخدامه. ترث الكتلة التعريفات المسماة من النطاق العام. إعادة تعريف اسم موجود في النطاق الحالي يعد خطأ.
تكوين = قيمة ;
التكوين يسمح بتغيير سلوك re2ocaml وتخصيص التعليمات البرمجية المولدة. للحصول على قائمة كاملة بالتكوينات التي يدعمها re2ocaml، راجع قسم التكوينات. اعتمادًا على تكوين معين، يمكن أن تكون القيمة كلمة مفتاحية، أو رقم صحيح غير سالب، أو سلسلة نصية من سطر واحد يجب أن تكون محاطة بعلامات اقتباس مزدوجة أو مفردة ما لم تتكون من أحرف أبجدية رقمية. ترث الكتلة التكوينات من النطاق العام ويمكنها إعادة تعريفها أو إضافة تكوينات جديدة. تؤثر التكوينات المعرفة داخل كتلة على الكتلة بأكملها، حتى لو ظهرت في نهايتها.
تعبير-عادي تعليمة برمجية
القاعدة تربط التعبير-العادي بإجراءها الدلالي (كتلة من التعليمات البرمجية في أقواس معقوفة، أو كتلة من التعليمات البرمجية تبدأ بـ := وتنتهي بسطر جديد متبوعًا بأي حرف غير مسافة بيضاء). إذا تطابق التعبير-العادي، يتم تنفيذ التعليمة البرمجية المرتبطة. إذا تطابقت قواعد متعددة، فإن أطول تطابق له الأسبقية. إذا تطابقت قواعد متعددة مع نفس السلسلة، فإن الأسبق له الأسبقية. هناك قاعدتان خاصتان: القاعدة الافتراضية * وقاعدة نهاية الإدخال $. يجب تعريف القاعدة الافتراضية دائمًا، ولها أقل أولوية بغض النظر عن مكانها في الكتلة، وتطابق أي وحدة رمز (ليس بالضرورة حرفًا صالحًا، راجع قسم دعم الترميز). يجب تعريف قاعدة نهاية الإدخال إذا تم استخدام طريقة معالجة نهاية الإدخال المقابلة. مع حالات البدء، تحتوي القواعد على صيغة أكثر تعقيدًا.
!تعليمة برمجية
الإجراء يربط كتلة تعليمات برمجية معرفة من قبل المستخدم بمكان معين في آلة الحالة المحدودة المولدة (بنفس طريقة ربط الإجراءات الدلالية بالتعليمات البرمجية للحالات النهائية). انظر قسم الإجراءات للحصول على قائمة كاملة بالإجراءات المحددة مسبقًا.
!توجيه ;
التوجيه هو أحد البيانات المحددة مسبقًا. لكل توجيه غرض فريد. انظر قسم التوجيهات للحصول على التفاصيل.

الكتل

علامات بداية ونهاية الكتلة هي إما /*!re2c و */، أو %{ و %} (كلا النمطين مدعومان). بدءًا من الإصدار 2.2، يمكن أن يكون للكتل أسماء اختيارية تسمح بالإشارة إليها في كتل أخرى. هناك أنواع مختلفة من الكتل:

/*!re2c[:<name>] ... */ أو %{[:<name>] ... %}
تحتوي الكتلة العامة على تعريفات وتكوينات وقواعد وتوجيهات. يقوم re2ocaml بترجمة التعبيرات العادية المرتبطة بكل قاعدة إلى آلة حالة محدودة حتمية، ويقوم بتشفيرها في شكل قفزات شرطية في اللغة المستهدفة ويستبدل الكتلة بالتعليمات البرمجية المولدة. تضاف الأسماء والتكوينات المعرفة في كتلة عامة إلى النطاق العام وتصبح مرئية للكتل اللاحقة. في بداية البرنامج، تتم تهيئة النطاق العام بـ خيارات سطر الأوامر.
/*!local:re2c[:<name>] ... */ أو %{local[:<name>] ... %}
الكتلة المحلية تشبه الكتلة العامة، ولكن الأسماء والتكوينات فيها لها نطاق محلي (لا تؤثر على الكتل الأخرى).
/*!rules:re2c[:<name>] ... */ أو %{rules[:<name>] ... %}
كتلة القواعد تشبه الكتلة المحلية، ولكنها لا تولد أي تعليمات برمجية بنفسها، ولا تضيف أي تعريفات إلى النطاق العام - فهي مخصصة لإعادة الاستخدام في كتل أخرى. هذه طريقة لمشاركة التعليمات البرمجية (مزيد من التفاصيل في قسم الكتل القابلة لإعادة الاستخدام). قبل الإصدار 2.2 من re2ocaml، تطلبت كتل القواعد الخيار -r --reusable.
/*!use:re2c[:<name>] ... */ أو %{use[:<name>] ... %}
كتلة استخدام تشير إلى كتلة قواعد معرفة مسبقًا. إذا تم تحديد الاسم، يبحث re2ocaml عن كتل قواعد بهذا الاسم. وإلا، يتم استخدام أحدث كتلة قواعد (سواء كانت مسماة أو غير مسماة). يمكن لكتلة الاستخدام إضافة تعريفاتها وتكويناتها وقواعدها الخاصة، والتي تضاف إلى تلك الخاصة بكتلة القواعد المشار إليها. قبل الإصدار 2.2 من re2ocaml، تطلبت كتل الاستخدام الخيار -r --reusable.
/*!max:re2c[:<name1>[:<name2>...]] ... */ أو %{max[:<name1>[:<name2>...]] ... %}
كتلة تولد تعريف YYMAXFILL. تحدد قائمة اختيارية بأسماء الكتل أي الكتل يجب تضمينها عند حساب قيمة YYMAXFILL (إذا كانت القائمة فارغة، يتم تضمين جميع الكتل). افتراضيًا، التعليمات البرمجية المولدة هي تعريف ماكرو لـ C (#define YYMAXFILL <n>)، أو متغير عام لـ Go (var YYMAXFILL int = <n>). يمكن تخصيصها بتكوين اختياري format يحدد سلسلة قالب يتم فيها استبدال @@{max} (أو @@ اختصارًا) بالقيمة الرقمية لـ YYMAXFILL.
/*!maxnmatch:re2c[:<name1>[:<name2>...]] ... */ أو %{maxnmatch[:<name1>[:<name2>...]] ... %}
كتلة تولد تعريف YYMAXNMATCH (يتطلب خيار -P --posix-captures). تحدد قائمة اختيارية بأسماء الكتل أي الكتل يجب تضمينها عند حساب Y

ملفات الصيغة

يعتمد دعم اللغات المختلفة في re2c على فكرة ملفات الصيغة. ملف الصيغة هو ملف تكوين يحدد صيغة اللغة المستهدفة - وليس اللغة بأكملها، ولكن جزءًا صغيرًا منها تستخدمه الشفرة التي تم إنشاؤها. تجعل ملفات الصيغة re2c مرنة للغاية، ولكن لا ينبغي استخدامها كبديل لتكوينات re2c:: الغرض منها هو تحديد صيغة اللغة المستهدفة، وليس تخصيص محلل لغوي معين. جميع اللغات المدعومة لديها ملفات صيغة افتراضية جزء من التوزيع (انظر الدليل الفرعي include/syntax)؛ وهي مضمنة أيضًا في الثنائي re2ocaml. يمكن للمستخدمين تقديم ملف صيغة مخصص يحل محل عدد قليل من التكوينات لواحدة من اللغات المدعومة، أو يمكنهم اختيار إعادة تعريف جميع التكوينات (في هذه الحالة يجب استخدام الخيار --lang none). تحتوي ملفات الصيغة على تكوينات من أربعة أنواع مختلفة: قوائم الميزات، وتكوينات اللغة، وتكوينات التضمين، وقوالب الشفرة.

قوائم الميزات

تحدد بعض تكوينات القوائم ميزات مختلفة تدعمها الواجهة الخلفية المعينة، بحيث يمكن لـ re2ocaml تقديم خطأ واضح إذا حاول المستخدم تمكين ميزة غير مدعومة:
supported_apis
قائمة بواجهات برمجة التطبيقات المدعومة مع عناصر محتملة simple، record، generic.
supported_api_styles
قائمة بأنماط واجهة برمجة التطبيقات المدعومة مع عناصر محتملة functions، free-form.
supported_code_models
قائمة بنماذج الشفرة المدعومة مع عناصر محتملة goto-label، loop-switch، recursive-functions.
supported_targets
قائمة بأهداف توليد الشفرة المدعومة مع عناصر محتملة code، dot، skeleton.
supported_features
قائمة بالميزات المدعومة مع عناصر محتملة nested-ifs، bitmaps، computed-gotos، case-ranges، monadic، unsafe، tags، captures، captvars.

backtick_quoted_strings
قيمة غير صفرية إذا كانت اللغة تحتوي على سلاسل محاطة بعلامات اقتباس خلفية.
single_quoted_strings
قيمة غير صفرية إذا كانت اللغة تحتوي على سلاسل محاطة بعلامات اقتباس مفردة.
indentation_sensitive
قيمة غير صفرية إذا كانت اللغة حساسة للمسافة البادئة.
wrap_blocks_in_braces
قيمة غير صفرية إذا كانت الكتل المركبة يجب أن تكون محاطة بأقواس متعرجة.

تكوينات التضمين

تحدد ملفات الصيغة القيم الأولية لجميع تكوينات re2c:، حيث يمكن أن تختلف للغات مختلفة. انظر قسم التكوينات للحصول على قائمة كاملة بجميع تكوينات التضمين ومعناها.

قوالب الشفرة

تحدد قوالب الشفرة صيغة اللغة المستهدفة. وهي مكتوبة بلغة بسيطة خاصة بالمجال مع النحو الرسمي التالي: 
code-template ::
      name '=' code-exprs ';'
    | CODE_TEMPLATE ';'
    | '<undefined>' ';'
code-exprs ::
      <EMPTY>
    | code-exprs code-expr
code-expr ::
      STRING
    | VARIABLE
    | optional
    | list
optional ::
      '(' CONDITIONAL '?' code-exprs ')'
    | '(' CONDITIONAL '?' code-exprs ':' code-exprs ')'
list ::
      '[' VARIABLE ':' code-exprs ']'
    | '[' VARIABLE '{' NUMBER '}' ':' code-exprs ']'
    | '[' VARIABLE '{' NUMBER ',' NUMBER '}' ':' code-exprs ']'

قالب الشفرة هو تسلسل من الثوابت النصية، والمتغيرات، والعناصر الاختيارية والقوائم، أو إشارة إلى قالب شفرة آخر، أو قيمة خاصة <undefined>. المتغيرات هي عناصر نائبة يتم استبدالها أثناء مرحلة توليد الشفرة. متغيرات القائمة خاصة: عند توسيع قوالب القائمة، يكرر re2ocaml تعبيرات الجانب الأيمن من العمود عدة مرات، ويستبدل في كل مرة تكرارات متغير القائمة بقيمة خاصة بهذه التكرارات. تحتوي القوائم على حدود اختيارية (القيم السالبة تُحسب من النهاية، على سبيل المثال -1 يعني العنصر الأخير). أسماء الشروط تبدأ بنقطة. قد تكون الشروط والمتغيرات محلية (خاصة بقالب الشفرة المحدد) أو عامة (مسموح بها في جميع قوالب الشفرة). عندما يقرأ re2ocaml ملف الصيغة، فإنه يتحقق من أن كل قالب شفرة يستخدم فقط المتغيرات والشروط المسموح بها فيه.

على سبيل المثال، يحدد قالب الشفرة التالي بنية "إذا كان ... إذن ... وإلا ..." للغة تشبه C:

Bounds checks are generated only in some states that are determined by the strongly connected components of the underlying automaton. Checks have the form (YYLIMIT - YYCURSOR) < n or YYLESSTHAN(n) with generic API, where n is the minimum number of characters that are needed for the lexer to proceed (it also means that the next bounds check will occur in at most n characters). If the check condition is true, the lexer has reached the end of input and will invoke YYFILL(n) that should either supply at least n input characters or not return. Custom checks

This example uses a custom end-of-input handling method based on generic API. The program counts space-separated single-quoted strings. It is the same as the sentinel example, except that the input is not null-terminated. To cover up for the absence of a sentinel character at the end of input, YYPEEK is redefined to perform a bounds check before it reads the next input character. This is inefficient because checks are done very often. If the check condition fails, YYPEEK returns the real character, otherwise it returns a fake sentinel character.

(* re2ocaml $INPUT -o $OUTPUT *)
type state = {
    str: string;
    mutable cur: int;
    lim: int;
}
(* expect a string without terminating null *)
%{
    re2c:api = generic;
    re2c:YYFN = ["lex;int", "st;state", "count;int"];
    re2c:YYPEEK = "if st.cur < st.lim then st.str.[st.cur] else '\\x00'";
    re2c:YYSKIP = "st.cur <- st.cur + 1;";
    re2c:yyfill:enable = 0;
    *      { -1 }
    [\x00] { count }
    [a-z]+ { lex st (count + 1) }
    [ ]+   { lex st count }
%}
let test(str, count) =
    let st = {str = str; cur = 0; lim = String.length str}
    in if not (lex st 0 = count) then raise (Failure "error")
let main () =
    test("", 0);
    test("one two three", 3);
    test("f0ur", -1)
let _ = main ()

BUFFER REFILLING

The need for buffering arises when the input cannot be mapped in memory all at once: either it is too large, or it comes in a streaming fashion (like reading from a socket). The usual technique in such cases is to allocate a fixed-sized memory buffer and process input in chunks that fit into the buffer. When the current chunk is processed, it is moved out and new data is moved in. In practice it is somewhat more complex, because lexer state consists not of a single input position, but a set of interrelated positions:

  • cursor: the next input character to be read (YYCURSOR in C pointer API or YYSKIP/YYPEEK in generic API)
  • limit: the position after the last available input character (YYLIMIT in C pointer API, implicitly handled by YYLESSTHAN in generic API)
  • marker: the position of the most recent match, if any (YYMARKER in default API or YYBACKUP/YYRESTORE in generic API)
  • token: the start of the current lexeme (implicit in re2ocaml API, as it is not needed for the normal lexer operation and can be defined and updated by the user)
  • context marker: the position of the trailing context (YYCTXMARKER in C pointer API or YYBACKUPCTX/YYRESTORECTX in generic API)
  • tag variables: submatch positions (defined with stags and mtags blocks and generic API primitives YYSTAGP/YYSTAGN/YYMTAGP/YYMTAGN)

Not all these are used in every case, but if used, they must be updated by YYFILL. All active positions are contained in the segment between token and cursor, therefore everything between buffer start and token can be discarded, the segment from token and up to limit should be moved to the beginning of buffer, and the free space at the end of buffer should be filled with new data. In order to avoid frequent YYFILL calls it is best to fill in as many input characters as possible (even though fewer characters might suffice to resume the lexer). The details of YYFILL implementation are slightly different depending on which EOF handling method is used: the case of EOF rule is somewhat simpler than the case of bounds-checking with padding. Also note that if -f --storable-state option is used, YYFILL has slightly different semantics (described in the section about storable state).

YYFILL with sentinel

If EOF rule is used, YYFILL is a function-like primitive that accepts no arguments and returns a value which is checked against zero. YYFILL invocation is triggered by condition YYLIMIT <= YYCURSOR in C pointer API and YYLESSTHAN() in generic API. A non-zero return value means that YYFILL has failed. A successful YYFILL call must supply at least one character and adjust input positions accordingly. Limit must always be set to one after the last input position in buffer, and the character at the limit position must be the sentinel symbol specified by re2c:eof configuration. (* re2ocaml $INPUT -o $OUTPUT *)
open Bytes
let bufsize = 4096
type state = {
    file: in_channel;
    yyinput: bytes;
    mutable yycursor: int;
    mutable yymarker: int;
    mutable yylimit: int;
    mutable token: int;
    mutable eof: bool;
}
type status = Ok | Eof | LongLexeme
let fill(st: state) : status =
    if st.eof then Eof else
    (* Error: lexeme too long. In real life could reallocate a larger buffer. *)
    if st.token < 1 then LongLexeme else (
    (* Shift buffer contents (discard everything up to the current token). *)
    blit st.yyinput st.token st.yyinput 0 (st.yylimit - st.token);
    st.yycursor <- st.yycursor - st.token;
    st.yymarker <- st.yymarker - st.token;
    st.yylimit <- st.yylimit - st.token;
    st.token <- 0;
    (* Fill free space at the end of buffer with new data from file. *)
    let n = input st.file st.yyinput st.yylimit (bufsize - st.yylimit - 1) in (* -1 for sentinel *)
    st.yylimit <- st.yylimit + n;
    if n = 0 then
        st.eof <- true; (* end of file *)
        set st.yyinput st.yylimit '\x00'; (* append sentinel *)
    Ok)

    re2c:YYFN = ["lex;int", "yyrecord;state", "count;int"];

    re2c:eof = 0;
    str = ['] ([^'\\] | [\\][^])* ['];
    *    { -1 }
    $    { count }
    str  { lex_loop yyrecord (count + 1) }
    [ ]+ { lex_loop yyrecord count }
%}
and lex_loop st count =
    st.token <- st.yycursor;
    lex st count

ملفات الترويسة

يسمح re2ocaml بتوليد ملف ترويسة من ملف الإدخال .re باستخدام الخيار --header أو تكوين re2c:header و أزواج الكتل من النوع /*!header:re2c:on*/ و /*!header:re2c:off*/، أو %{header:on%} و %{header:off%}. الكتلة الأولى تحدد بداية ملف الترويسة، والكتلة الثانية تحدد نهايته. يتم معالجة كل شيء بين هاتين الكتلتين بواسطة re2ocaml، ويتم كتابة الكود الذي تم إنشاؤه في الملف المحدد باستخدام الخيار --header أو تكوين re2c:header (أو stdout إذا لم يتم استخدام أي خيار أو تكوين). قد تكون ملفات الترويسة المولدة تلقائيًا مطلوبة في الحالات التي يتم فيها استخدام re2ocaml لتوليد تعريفات يجب أن تكون مرئية من وحدات الترجمة الأخرى.

إليك مثال لتوليد ملف ترويسة يحتوي على تعريف لحالة محلل الأكواد مع متغيرات العلامات (عدد المتغيرات يعتمد على قواعد اللغة المنتظمة وهو غير معروف للمبرمج).

برامج الهيكل العظمي

باستخدام الخيار -S, --skeleton، يتجاهل re2ocaml جميع الأكواد غير re2ocaml وينشئ برنامج C مستقل يمكن تجميعه وتنفيذه لاحقًا. يتكون البرنامج من كود محلل الأكواد وبيانات الإدخال. لكل DFA تم إنشاؤه (كتلة أو شرط)، يقوم re2ocaml بإنشاء محلل أكواد مستقل وملفين: ملف .input يحتوي على سلاسل مشتقة من DFA وملف .keys يحتوي على نتائج المطابقة المتوقعة. يقوم البرنامج بتشغيل كل محلل أكواد على ملف .input المقابل له ويقارن النتائج بالتوقعات. برامج الهيكل العظمي مفيدة جدًا لعدد من الأسباب:

  • يمكنها التحقق من صحة تحسينات re2ocaml المختلفة (يتم إنشاء البيانات في مرحلة مبكرة من العملية، قبل إجراء أي تحويلات DFA).
  • قد يكون توليد مجموعة من بيانات الإدخال مع تغطية جيدة مفيدًا لكل من الاختبار وقياس الأداء.
  • يسمح توليد برامج قابلة للتنفيذ مستقلة بالحصول على حالات اختبار مصغرة (قد يكون الكود الأصلي كبيرًا أو به الكثير من الاعتماديات).

الصعوبة في توليد بيانات الإدخال هي أنه بالنسبة لجميع الحالات باستثناء الحالات البسيطة جدًا، يكون عدد سلاسل الإدخال الممكنة كبيرًا جدًا (حتى لو كان طول السلسلة محدودًا). يحل re2ocaml هذه الصعوبة عن طريق توليد عدد كافٍ من السلاسل لتغطية جميع انتقالات DFA تقريبًا. يستخدم الخوارزمية التالية. أولاً، يقوم ببناء هيكل عظمي لـ DFA. بالنسبة للتشفيرات بحجم وحدات كود 1 بايت (مثل ASCII، UTF-8، و EBCDIC) يكون الهيكل العظمي مجرد نسخة طبق الأصل من DFA الأصلي. بالنسبة للتشفيرات بوحدات كود متعددة البايت، يكون الهيكل العظمي نسخة من DFA مع إزالة انتقالات معينة: على وجه التحديد، يأخذ re2ocaml بحد أقصى 256 وحدة كود لكل نطاق مستمر منفصل يتوافق مع انتقال DFA. القيم المختارة موزعة بالتساوي وتشمل حدود النطاق. بدلاً من محاولة تغطية جميع المسارات الممكنة في الهيكل العظمي (وهو أمر غير ممكن)، يقوم re2ocaml بتوليد عدد كافٍ من المسارات لتغطية جميع انتقالات الهيكل العظمي، وبالتالي تشغيل القفزات الشرطية المقابلة في محلل الأكواد. التنفيذ الخوارزمي محدود بحوالي 1 جيجابايت من الانتقالات ويستهلك كمية ثابتة من الذاكرة (يقوم re2ocaml بكتابة البيانات إلى الملف بمجرد إنشائها).

التصور والتصحيح

باستخدام الخيار -D, --emit-dot، لا يقوم re2ocaml بتوليد كود. بدلاً من ذلك، يقوم بطرح DFA الذي تم إنشاؤه بتنسيق DOT. يمكن تحويل هذا الطرح إلى صورة لـ DFA باستخدام Graphviz أو مكتبة أخرى. لاحظ أن هذا الخيار يعرض DFA النهائي بعد أن مر بعدد من التحسينات والتحويلات. يمكن طرح المراحل المبكرة باستخدام خيارات تصحيح مختلفة، مثل --dump-nfa، --dump-dfa-raw، إلخ (انظر القائمة الكاملة للخيارات).
