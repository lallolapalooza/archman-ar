restic-key-remove(1)

الحزمة: restic

التراخيص: BSD

restic-key-remove(1) restic-key-remove(1)

الاسم

restic-key-remove - إزالة معرف المفتاح (كلمة المرور) من المستودع.

ملخص

restic key remove [ID] [flags]

الوصف

الأمر الفرعي "remove" يقوم بإزالة معرف المفتاح المحدد. الأمر "remove" لا يسمح بإزالة المفتاح الحالي المستخدم للوصول إلى المستودع.

حالة الخروج

حالة الخروج هي 0 إذا كان الأمر ناجحًا. حالة الخروج هي 1 إذا كان هناك أي خطأ. حالة الخروج هي 10 إذا كان المستودع غير موجود. حالة الخروج هي 11 إذا كان المستودع مقفلاً بالفعل. حالة الخروج هي 12 إذا كانت كلمة المرور غير صحيحة.

الخيارات

-h, --help[=false] تعليمات لـ remove

خيارات موروثة من الأوامر الأصل

--cacert=[] ملف لتحميل شهادات الجذر منه (افتراضي: استخدام شهادات النظام أو $RESTIC_CACERT)

--cache-dir="" تعيين دليل ذاكرة التخزين المؤقت. (افتراضي: استخدام دليل ذاكرة التخزين المؤقت الافتراضي للنظام)

--cleanup-cache[=false] إزالة أدلة ذاكرة التخزين المؤقت القديمة تلقائيًا

--compression=auto وضع الضغط (متوفر فقط لتنسيق المستودع الإصدار 2)، أحد الخيارات (auto|off|max) (افتراضي: $RESTIC_COMPRESSION)

--http-user-agent="" تعيين وكيل مستخدم HTTP لطلبات HTTP الصادرة

--insecure-no-password[=false] استخدام كلمة مرور فارغة للمستودع، يجب تمريرها إلى كل أمر restic (غير آمن)

--insecure-tls[=false] تخطي التحقق من شهادة TLS عند الاتصال بالمستودع (غير آمن)

--json[=false] تعيين وضع الإخراج إلى JSON للأوامر التي تدعمه

--key-hint="" معرف المفتاح لمحاولة فك تشفيره أولاً (افتراضي: $RESTIC_KEY_HINT)

--limit-download=0 يحد التنزيلات إلى معدل أقصى بالكيلوبايت/ثانية. (افتراضي: غير محدود)

--limit-upload=0 يحد التحميلات إلى معدل أقصى بالكيلوبايت/ثانية. (افتراضي: غير محدود)

--no-cache[=false] عدم استخدام ذاكرة تخزين مؤقت محلية

--no-extra-verify[=false] تخطي التحقق الإضافي من البيانات قبل التحميل (انظر التوثيق)

--no-lock[=false] عدم قفل المستودع، هذا يسمح ببعض العمليات على المستودعات للقراءة فقط

-o, --option=[] تعيين خيار ممتد (key=value، يمكن تحديده عدة مرات)

--pack-size=0 تعيين حجم الحزمة المستهدف بالميجابايت، قد تكون ملفات الحزمة المنشأة أكبر (افتراضي: $RESTIC_PACK_SIZE)

--password-command="" أمر shell للحصول على كلمة مرور المستودع منه (افتراضي: $RESTIC_PASSWORD_COMMAND)

-p, --password-file="" ملف للقراءة منه كلمة مرور المستودع (افتراضي: $RESTIC_PASSWORD_FILE)

-q, --quiet[=false] عدم عرض تقرير تقدم شامل

-r, --repo="" المستودع للنسخ الاحتياطي إليه أو الاستعادة منه (افتراضي: $RESTIC_REPOSITORY)

--repository-file="" ملف لقراءة موقع المستودع منه (افتراضي: $RESTIC_REPOSITORY_FILE)

--retry-lock=0s إعادة محاولة قفل المستودع إذا كان مقفلاً بالفعل، يأخذ قيمة مثل 5m أو 2h (افتراضي: لا توجد إعادة محاولات)

--stuck-request-timeout=5m0s مدة بعد إعادة محاولة الطلبات العالقة

--tls-client-cert="" مسار إلى ملف يحتوي على شهادة عميل TLS بترميز PEM والمفتاح الخاص (افتراضي: $RESTIC_TLS_CLIENT_CERT)

-v, --verbose[=0] كن مفصلاً (حدد عدة مرات أو مستوى باستخدام --verbose=n``، الحد الأقصى للمستوى/المرات هو 2)
