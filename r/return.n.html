return(n)

الحزمة: tcl

التراخيص: TCL

في البرنامج النصي. أي رمز إرجاع استثنائي (غير TCL_OK) تم إرجاعه من تقييم أمر ما يؤدي إلى مقاطعة التدفق إلى الأمر التالي. يتوقف تقييم البرنامج النصي، ويصبح رمز الإرجاع الاستثنائي من الأمر هو رمز الإرجاع للتقييم الكامل للبرنامج النصي. هذه هي الآلية التي بموجبها تسبب الأخطاء أثناء تقييم البرنامج النصي في مقاطعة وتفريغ مكدس الاستدعاءات. وهي أيضًا الآلية التي بموجبها تسبب أوامر مثل break، و continue، و return في إنهاء تقييم البرنامج النصي دون تقييم جميع الأوامر بالتتابع.

تقوم بعض أوامر Tcl المضمنة بتقييم البرامج النصية كجزء من عملها. يمكن لهذه الأوامر الاستفادة من رموز الإرجاع الاستثنائية لتمكين ميزات خاصة. على سبيل المثال، أوامر Tcl المضمنة التي توفر حلقات – مثل while، و for، و foreach – تقوم بتقييم برنامج نصي هو جسم الحلقة. إذا كان تقييم جسم الحلقة يرجع رمز الإرجاع TCL_BREAK أو TCL_CONTINUE، يمكن لأمر الحلقة التفاعل بطريقة تعطي أوامر break و continue تفسيرها الموثق في الحلقات.

يتضمن استدعاء الإجراء أيضًا تقييم برنامج نصي، وهو جسم الإجراء. يوفر استدعاء الإجراء معاملة خاصة عند تقييم جسم الإجراء يرجع رمز الإرجاع TCL_RETURN. في تلك الظروف، يتم إنقاص مدخل -level في قاموس خيارات الإرجاع. إذا كان، بعد الإنقاص، قيمة مدخل -level هي 0، فإن قيمة مدخل -code تصبح رمز الإرجاع للإجراء. إذا كان، بعد الإنقاص، قيمة مدخل -level أكبر من صفر، فإن رمز الإرجاع للإجراء هو TCL_RETURN. إذا حدث استدعاء الإجراء أثناء تقييم جسم إجراء آخر، فستتكرر العملية صعودًا في مكدس الاستدعاءات، مع إنقاص قيمة مدخل -level في كل مستوى، بحيث يكون *الكود* هو رمز الإرجاع لمستوى الأمر الحالي *level* مستويات أعلى في مكدس الاستدعاءات. ينفذ الأمر source نفس معالجة رمز الإرجاع TCL_RETURN، وهذا يفسر تشابه استدعاء return أثناء source مع استدعاء return داخل إجراء.

رمز الإرجاع لأمر return نفسه يؤدي إلى هذه المعالجة الخاصة من خلال استدعاء الإجراء. إذا تم تزويد return بالخيار -level 0، فإن رمز الإرجاع لأمر return نفسه سيكون القيمة *code* للخيار -code (أو TCL_OK افتراضيًا). أي قيمة أخرى للخيار -level (بما في ذلك القيمة الافتراضية 1) ستؤدي إلى أن يكون رمز الإرجاع لأمر return نفسه هو TCL_RETURN، مما يؤدي إلى العودة من الإجراء المحيط.

أمثلة

أولاً، مثال بسيط على استخدام return للعودة من إجراء، مما يؤدي إلى مقاطعة جسم الإجراء.

proc printOneLine {} {
    puts "line 1"    ;# This line will be printed.
    return
    puts "line 2"    ;# This line will not be printed.
}

بعد ذلك، مثال على استخدام return لتعيين القيمة التي يرجعها الإجراء.

proc returnX {} {return X}
puts [returnX]    ;# prints "X"

بعد ذلك، مثال أكثر اكتمالاً، باستخدام return -code error للإبلاغ عن وسائط غير صالحة.

proc factorial {n} {
    if {![string is integer $n] || ($n < 0)} {
        return -code error \
                "expected non-negative integer,\
                but got \"$n\""
    }
    if {$n < 2} {
        return 1
    }
    set m [expr {$n - 1}]
    set code [catch {factorial $m} factor]
    if {$code != 0} {
        return -code $code $factor
    }
    set product [expr {$n * $factor}]
    if {$product < 0} {
        return -code error \
                "overflow computing factorial of $n"
    }
    return $product
}

بعد ذلك، إجراء بديل لأمر break.

proc myBreak {} {
    return -code break
}

مع الخيار -level 0، يمكن أن يعمل return نفسه كبديل لـ break، بمساعدة interp alias.

interp alias {} Break {} return -level 0 -code break

مثال على استخدام catch و return -options لإعادة إثارة خطأ تم التقاطه:

proc doSomething {} {
    set resource [allocate]
    catch {
        # Long script of operations
        # that might raise an error
    } result options
    deallocate $resource
    return -options $options $result
}

أخيرًا، مثال على استخدام متقدم لخيارات return لإنشاء إجراء بديل لأمر return نفسه:

proc myReturn {args} {
    set result ""
    if {[llength $args] % 2} {
        set result [lindex $args end]
        set args [lrange $args 0 end-1]
    }
    set options [dict merge {-level 1} $args]
    dict incr options -level
    return -options $options $result
}
