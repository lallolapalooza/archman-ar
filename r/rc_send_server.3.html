rc_send_server(3)

الحزمة: radcli

التراخيص: BSD

rc_send_server(3) مكتبة عميل Radius rc_send_server(3)

الاسم

rc_send_server -

الخلاصة

#include <radcli/radcli.h>

int rc_send_server(
    rc_handle  *rh,
    SEND_DATA  *data,
    char       *msg,
    rc_type     type
);

الوصف

يرسل طلبًا إلى خادم RADIUS وينتظر الرد

المعلمات

rh
مقبض للإعدادات المحللة
data
مؤشر إلى هيكل SEND_DATA
msg
يجب أن يكون مصفوفة بحجم PW_MAX_MSG_SIZE أو NULL؛ سيحتوي على تسلسل أي PW_REPLY_MESSAGE مستلم.
type
يجب أن يكون AUTH أو ACCT

الهياكل


struct send_data {
  uint8_t      code;          // كود حزمة RADIUS. 
  uint8_t      seq_nbr;       // رقم تسلسل الحزمة. 
  char        *server;        // اسم/عنوان خادم RADIUS. 
  int          svc_port;      // منفذ وجه بروتوكول RADIUS. 
  char        *secret;        // السر المشترك لخادم RADIUS. 
  int          timeout;       // مهلة الجلسة بالثواني. 
  int          retries;
  VALUE_PAIR  *send_pairs;    // المزيد من أزواج القيمة/السمة للإرسال. 
  VALUE_PAIR  *receive_pairs; // حيث يتم وضع أزواج القيمة/السمة المستلمة. 
};

قيمة الإرجاع

OK_RC (0) عند النجاح، TIMEOUT_RC عند انتهاء المهلة REJECT_RC عند رفض الوصول، أو قيمة سالبة عند الفشل كقيمة إرجاع.
