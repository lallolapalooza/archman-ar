rc_buildreq(3)

الحزمة: radcli

التراخيص: BSD

rc_buildreq(3) مكتبة عميل Radius rc_buildreq(3)

الاسم

rc_buildreq -

الملخص

#include <radcli/radcli.h>

void rc_buildreq(
    rc_handle const  *rh,
    SEND_DATA        *data,
    int               code,
    char             *server,
    unsigned short    port,
    char             *secret,
    int               timeout,
    int               retries
);

الوصف

بناء طلب RADIUS هيكلي باستخدام معلومات من ملف التكوين

المعلمات

rh
مقبض للتكوين المحلل.
data
مؤشر إلى هيكل SEND_DATA.
code
واحد من أكواد RADIUS القياسية (على سبيل المثال، PW_ACCESS_REQUEST).
server
اسم الخادم.
port
رقم منفذ الخادم.
secret
السر المستخدم من قبل الخادم.
timeout
المهلة بالثواني للرسالة.
retries
عدد المحاولات.

الهياكل


struct send_data {
  uint8_t      code;          // كود حزمة RADIUS. 
  uint8_t      seq_nbr;       // رقم تسلسل الحزمة. 
  char        *server;        // اسم/عنوان خادم RADIUS. 
  int          svc_port;      // منفذ وجهة بروتوكول RADIUS. 
  char        *secret;        // السر المشترك لخادم RADIUS. 
  int          timeout;       // مهلة الجلسة بالثواني. 
  int          retries;
  VALUE_PAIR  *send_pairs;    // المزيد من أزواج المفتاح/القيمة للإرسال. 
  VALUE_PAIR  *receive_pairs; // مكان وضع أزواج المفتاح/القيمة المستلمة. 
};
