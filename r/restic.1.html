restic(1)

الحزمة: restic

التراخيص: BSD

restic backup(1) restic backup(1)

الاسم

restic - النسخ الاحتياطي واستعادة الملفات

الملخص

restic [علامات]

الوصف

restic هو برنامج نسخ احتياطي يسمح بحفظ إصدارات متعددة من الملفات والمجلدات في مستودع مشفر مخزن على أنظمة خلفية مختلفة.

يمكن العثور على الوثائق الكاملة على https://restic.readthedocs.io/ .

خيارات

--cacert=[] ملف لتحميل شهادات الجذر منه (الافتراضي: استخدام شهادات النظام أو $RESTIC_CACERT)

--cache-dir="" تعيين دليل ذاكرة التخزين المؤقت. (الافتراضي: استخدام دليل ذاكرة التخزين المؤقت الافتراضي للنظام)

--cleanup-cache[=false] إزالة أدلة ذاكرة التخزين المؤقت القديمة تلقائيًا

--compression=auto وضع الضغط (متوفر فقط لإصدار تنسيق المستودع 2)، واحد من (auto|off|max) (الافتراضي: $RESTIC_COMPRESSION)

-h, --help[=false] المساعدة لـ restic

--http-user-agent="" تعيين وكيل مستخدم HTTP لطلبات HTTP الصادرة

--insecure-no-password[=false] استخدام كلمة مرور فارغة للمستودع، يجب تمريرها مع كل أمر restic (غير آمن)

--insecure-tls[=false] تخطي التحقق من صحة شهادة TLS عند الاتصال بالمستودع (غير آمن)

--json[=false] تعيين وضع الإخراج إلى JSON للأوامر التي تدعمه

--key-hint="" معرف المفتاح للمفتاح لمحاولة فك تشفيره أولاً (الافتراضي: $RESTIC_KEY_HINT)

--limit-download=0 يحد التنزيلات إلى معدل أقصى بالكيلوبايت/ثانية. (الافتراضي: غير محدود)

--limit-upload=0 يحد التحميلات إلى معدل أقصى بالكيلوبايت/ثانية. (الافتراضي: غير محدود)

--no-cache[=false] عدم استخدام ذاكرة تخزين مؤقت محلية

--no-extra-verify[=false] تخطي التحقق الإضافي من البيانات قبل التحميل (انظر الوثائق)

--no-lock[=false] عدم قفل المستودع، يسمح هذا ببعض العمليات على المستودعات للقراءة فقط

-o, --option=[] تعيين خيار ممتد (key=value، يمكن تحديده عدة مرات)

--pack-size=0 تعيين حجم الحزمة المستهدف بالميغابايت، قد تكون ملفات الحزمة المنشأة أكبر (الافتراضي: $RESTIC_PACK_SIZE)

--password-command="" أمر shell للحصول على كلمة مرور المستودع منه (الافتراضي: $RESTIC_PASSWORD_COMMAND)

-p, --password-file="" ملف للقراءة منه كلمة مرور المستودع (الافتراضي: $RESTIC_PASSWORD_FILE)

-q, --quiet[=false] عدم عرض تقرير تقدم شامل

-r, --repo="" المستودع للنسخ الاحتياطي إليه أو الاستعادة منه (الافتراضي: $RESTIC_REPOSITORY)

--repository-file="" ملف للقراءة منه موقع المستودع (الافتراضي: $RESTIC_REPOSITORY_FILE)

--retry-lock=0s إعادة المحاولة لقفل المستودع إذا كان مقفلاً بالفعل، يأخذ قيمة مثل 5m أو 2h (الافتراضي: لا توجد محاولات إعادة)

--stuck-request-timeout=5m0s مدة زمنية سيتم بعدها إعادة محاولة الطلبات العالقة

--tls-client-cert="" مسار إلى ملف يحتوي على شهادة عميل TLS مشفرة بصيغة PEM ومفتاح خاص (الافتراضي: $RESTIC_TLS_CLIENT_CERT)

-v, --verbose[=0] كن مفصلاً (حدد عدة مرات أو مستوى باستخدام --verbose=n``، أقصى مستوى/مرات هو 2)
