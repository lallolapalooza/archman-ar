المكتبة

مكتبة C القياسية (libc).

الملخص

#include <sys/types.h>
#include <sys/socket.h>
#include <netinet/in.h>

int ruserok(const char *rhost, int rport, const char *luser);
int ruserok_af(const char *rhost, int rport, const char *luser, int domain);
int iruserok(const char *rhost, int rport, const char *luser);
int iruserok_af(const char *rhost, int rport, const char *luser, int domain);
int rresvport(int *alport);
int rresvport_af(int *alport, int domain);

الوصف

هذه الدوال تستخدم للتحقق من صلاحيات الوصول للمستخدمين عن بعد. تستخدم في بروتوكولات مثل rlogin و rsh. تسمح هذه الدوال للتطبيق الخادم بالتحقق مما إذا كان المستخدم الذي يطلب الاتصال من مضيف بعيد (rhost) هو المستخدم المحلي (luser) المسموح له بالوصول.

ruserok()
الدالة الأصلية للتحقق من اسم مضيف بعيد.

الوسيطات:

rhost
اسم المضيف البعيد.
rport
رقم المنفذ البعيد.
luser
اسم المستخدم المحلي الذي يرغب في التحقق من صلاحياته.
iruserok()
متشابهة مع ruserok ولكنها لا تقوم بتعيين اسم المضيف البعيد بشكل صريح، وغالبًا ما تعتمد على تعيينات أخرى. تم تقديمها في glibc 2.2.
ruserok_af()
متشابهة مع ruserok ولكنها تدعم نطاقات عناوين مختلفة (مثل IPv4 و IPv6) عن طريق الوسيط domain.
iruserok_af()
متشابهة مع iruserok ولكنها تدعم نطاقات عناوين مختلفة.
rresvport()
تستخدم لتعيين منفذ مصدر عشوائي في النطاق غير المميز (unprivileged) للاتصالات الصادرة.
rresvport_af()
متشابهة مع rresvport ولكنها تدعم نطاقات عناوين مختلفة.

قيمة الإرجاع

تُرجع هذه الدوال 0 عند النجاح، و -1 عند الفشل، مع تعيين errno بشكل مناسب.

السمات

للحصول على شرح للمصطلحات المستخدمة في هذا القسم، راجع attributes(7).

ruserok_af()
سمات، تعاريف، واستخدامات في threads(7).
rresvport_af()
سمات، تعاريف، واستخدامات في threads(7).
ruserok_af()
ND_NOMEMP، ND_ALTBASE
rresvport_af()
ND_NOMEMP، ND_ALTBASE

المعايير

تم تقديم الدوال ذات اللاحقة "_af" في glibc 2.2. الدوال بدون اللاحقة "_af" أقدم. النسخ الأصلية لهذه الدوال قد لا تكون متاحة على جميع الأنظمة.

التاريخ

تم تقديم الدوال ruserok و rresvport في إصدارات BSD المبكرة. النسخ ذات اللاحقة "_af" هي إضافات أحدث.

rresvport_af()
glibc 2.2.
ruserok_af()
glibc 2.2.

Solaris, 4.2BSD. النسخ ذات اللاحقة "_af" هي إضافات أحدث، وليست موجودة على نطاق واسع من الأنظمة.

الأخطاء

تم الإعلان عن iruserok() و iruserok_af() في ملفات ترويسة glibc فقط منذ glibc 2.12.
