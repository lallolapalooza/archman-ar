rand(3) دليل الدوال المكتبية rand(3)

الاسم

rand, rand_r, srand - مولد أرقام عشوائية زائفة

المكتبة

مكتبة C القياسية (libc, -lc)

الملخص

#include <stdlib.h>
int rand(void);
void srand(unsigned int seed);
[[deprecated]] int rand_r(unsigned int *seedp);

الوصف

الدالة rand() تعيد عددًا صحيحًا عشوائيًا زائفًا في النطاق من 0 إلى RAND_MAX شاملًا (أي، النطاق الرياضي [0, RAND_MAX]).

الدالة srand() تعين وسيطتها كبذرة لسلسلة جديدة من الأعداد العشوائية الزائفة التي تعيدها الدالة rand(). يمكن تكرار هذه السلاسل عن طريق استدعاء srand() بنفس قيمة البذرة.

إذا لم يتم توفير قيمة بذرة، فإن الدالة rand() يتم تهيئتها تلقائيًا بقيمة 1.

الدالة rand() ليست قابلة لإعادة الدخول (reentrant)، لأنها تستخدم حالة مخفية يتم تعديلها عند كل استدعاء. قد تكون هذه مجرد قيمة البذرة لاستخدامها في الاستدعاء التالي، أو قد تكون شيئًا أكثر تعقيدًا. للحصول على سلوك قابل للتكرار في تطبيق متعدد الخيوط، يجب توضيح هذه الحالة؛ يمكن القيام بذلك باستخدام الدالة القابلة لإعادة الدخول rand_r().

مثل rand()، تعيد rand_r() عددًا صحيحًا عشوائيًا زائفًا في النطاق [0, RAND_MAX]. الوسيطة seedp هي مؤشر إلى unsigned int يستخدم لتخزين الحالة بين الاستدعاءات. إذا تم استدعاء rand_r() بنفس القيمة الأولية للعدد الصحيح المشار إليه بواسطة seedp، ولم يتم تعديل هذه القيمة بين الاستدعاءات، فستنتج نفس التسلسل العشوائي الزائف.

القيمة المشار إليها بواسطة الوسيطة seedp للدالة rand_r() توفر فقط كمية صغيرة جدًا من الحالة، لذا ستكون هذه الدالة مولدًا عشوائيًا زائفًا ضعيفًا. جرب drand48_r(3) بدلاً من ذلك.

قيمة الإرجاع

الدالتان rand() و rand_r() تعيدان قيمة بين 0 و RAND_MAX (شاملة). الدالة srand() لا تعيد أي قيمة.

الإصدارات

تستخدم إصدارات rand() و srand() في مكتبة Linux C نفس مولد الأرقام العشوائية مثل random(3) و srandom(3)، لذا يجب أن تكون البتات ذات الرتب المنخفضة عشوائية مثل البتات ذات الرتب العالية. ومع ذلك، في تطبيقات rand() القديمة، وفي التطبيقات الحالية على أنظمة مختلفة، تكون البتات ذات الرتب المنخفضة أقل عشوائية بكثير من البتات ذات الرتب العالية. لا تستخدم هذه الدالة في التطبيقات المخصصة للنقل عندما تكون العشوائية الجيدة مطلوبة. (استخدم random(3) بدلاً من ذلك.)

أمثلة

يقدم POSIX.1-2001 المثال التالي لتطبيق rand() و srand()، وقد يكون مفيدًا عندما يحتاج المرء إلى نفس التسلسل على جهازي كمبيوتر مختلفين.


static unsigned long next = 1;
/* RAND_MAX مفترض أنها 32767 */
int myrand(void) {
    next = next * 1103515245 + 12345;
    return((unsigned) (next/65536) % 32768);
}
void mysrand(unsigned int seed) {
    next = seed;
}

يمكن استخدام البرنامج التالي لعرض التسلسل العشوائي الزائف الذي تنتجه rand() عند إعطائه بذرة معينة. عندما تكون البذرة -1، يستخدم البرنامج بذرة عشوائية.


#include <stdio.h>
#include <stdlib.h>
int
main(int argc, char *argv[])
{
    int           r;
    unsigned int  seed, nloops;
    if (argc != 3) {
        fprintf(stderr, "Usage: %s <seed> <nloops>\n", argv[0]);
        exit(EXIT_FAILURE);
    }
    seed = atoi(argv[1]);
    nloops = atoi(argv[2]);
    if (seed == -1) {
        seed = arc4random();
        printf("seed: %u\n", seed);
    }
    srand(seed);
    for (unsigned int j = 0; j < nloops; j++) {
        r =  rand();
        printf("%d\n", r);
    }
    exit(EXIT_SUCCESS);
}
