recv(2) دليل استدعاءات النظام recv(2)

الاسم

recv، recvfrom، recvmsg — استقبال رسالة من مقبس

المكتبة

مكتبة C القياسية (libc، -lc)

الملخص

#include <sys/socket.h>
ssize_t recv(int sockfd, void buf[size], size_t size,
                 int flags);
ssize_t recvfrom(int sockfd, void buf[restrict size], size_t size,
                 int flags,
                 struct sockaddr *_Nullable restrict src_addr,
                 socklen_t *_Nullable restrict addrlen);
ssize_t recvmsg(int sockfd, struct msghdr *msg, int flags);

الوصف

تُستخدم استدعاءات recv() و recvfrom() و recvmsg() لاستقبال رسائل من مقبس. يمكن استخدامها لاستقبال البيانات على المقابس المتصلة وغير المتصلة. هذه الصفحة تصف أولاً الميزات المشتركة لجميع استدعاءات النظام الثلاثة، ثم تصف الاختلافات بين الاستدعاءات.

الفرق الوحيد بين recv() و read(2) هو وجود flags. مع قيمة flags صفر، فإن recv() مكافئة بشكل عام لـ read(2) (ولكن انظر الملاحظات). أيضًا، الاستدعاء التالي


recv(sockfd, buf, size, flags);

مكافئ لـ


recvfrom(sockfd, buf, size, flags, NULL, NULL);

ترجع جميع الاستدعاءات الثلاثة حجم الرسالة عند اكتمالها بنجاح. إذا كانت الرسالة طويلة جدًا بحيث لا تناسب المخزن المؤقت المقدم، فقد يتم التخلص من البايتات الزائدة اعتمادًا على نوع المقبس الذي تم استقبال الرسالة منه.

إذا لم تكن هناك رسائل متاحة على المقبس، تنتظر استدعاءات الاستقبال وصول رسالة، إلا إذا كان المقبس غير متزامن (انظر fcntl(2))، وفي هذه الحالة يتم إرجاع القيمة -1 ويتم تعيين errno إلى EAGAIN أو EWOULDBLOCK. عادةً ما ترجع استدعاءات الاستقبال أي بيانات متاحة، حتى الحد الأقصى للمبلغ المطلوب، بدلاً من الانتظار لاستلام المبلغ الكامل المطلوب.

يمكن للتطبيق استخدام select(2) أو poll(2) أو epoll(7) لتحديد متى تصل بيانات إضافية على المقبس.

وسيطة flags

تتكون وسيطة flags عن طريق ORing قيمة واحدة أو أكثر من القيم التالية:

MSG_CMSG_CLOEXEC (recvmsg() فقط؛ منذ Linux 2.6.23)
تعيين علامة الإغلاق عند التنفيذ لموصّل الملف المستلم عبر موصل ملف خاص بنطاق UNIX باستخدام العملية SCM_RIGHTS (موصوفة في unix(7)). هذه العلامة مفيدة لنفس الأسباب التي توجد بها علامة O_CLOEXEC الخاصة بـ open(2).
MSG_DONTWAIT (منذ Linux 2.2)
تمكين التشغيل غير المتزامن؛ إذا كانت العملية ستتوقف، يفشل الاستدعاء مع EAGAIN أو EWOULDBLOCK. هذا يوفر سلوكًا مشابهًا لتعيين علامة O_NONBLOCK (عبر عملية F_SETFL الخاصة بـ fcntl(2))، ولكنه يختلف في أن MSG_DONTWAIT هي خيار لكل استدعاء، بينما O_NONBLOCK هو إعداد على وصف الملف المفتوح (انظر open(2))، والذي سيؤثر على جميع الخيوط في العملية المستدعية بالإضافة إلى العمليات الأخرى التي تشير إلى نفس وصف الملف المفتوح.
MSG_ERRQUEUE (منذ Linux 2.2)
تحدد هذه العلامة أن الأخطاء المخزنة مؤقتًا يجب استقبالها من قائمة انتظار أخطاء المقبس. يتم تمرير الخطأ في رسالة إضافية بنوع يعتمد على البروتوكول (بالنسبة لـ IPv4 IP_RECVERR). يجب على المستخدم توفير مخزن مؤقت بحجم كافٍ. انظر cmsg(3) و ip(7) لمزيد من المعلومات. يتم تمرير حمولة الحزمة الأصلية التي تسببت في الخطأ كبيانات عادية عبر msg_iovec. يتم توفير عنوان الوجهة الأصلي للحزمة التي سببت الخطأ عبر msg_name.
يتم توفير الخطأ في بنية sock_extended_err:

#define SO_EE_ORIGIN_NONE    0
#define SO_EE_ORIGIN_LOCAL   1
#define SO_EE_ORIGIN_ICMP    2
#define SO_EE_ORIGIN_ICMP6   3
struct sock_extended_err
{
    uint32_t ee_errno;   /* رقم الخطأ */
    uint8_t  ee_origin;  /* من أين نشأ الخطأ */
    uint8_t  ee_type;    /* النوع */
    uint8_t  ee_code;    /* الرمز */
    uint8_t  ee_pad;     /* حشو */
    uint32_t ee_info;    /* معلومات إضافية */
    uint32_t ee_data;    /* بيانات أخرى */
    /* قد يتبعها المزيد من البيانات */
};
struct sockaddr *SO_EE_OFFENDER(struct sock_extended_err *);

يحتوي ee_errno على رقم errno الخاص بالخطأ المخزن. ee_origin هو رمز مصدر الخطأ. الحقول الأخرى خاصة بالبروتوكول. الماكرو SO_EE_OFFENDER يرجع مؤشرًا إلى عنوان الكائن الشبكي الذي نشأ منه الخطأ بالنظر إلى مؤشر إلى الرسالة الإضافية. إذا لم يكن هذا العنوان معروفًا، فإن عضو sa_family الخاص بـ sockaddr يحتوي على AF_UNSPEC والحقول الأخرى لـ sockaddr غير محددة. يتم تمرير حمولة الحزمة التي سببت الخطأ كبيانات عادية.
بالنسبة للأخطاء المحلية، لا يتم تمرير عنوان (يمكن التحقق من ذلك باستخدام عضو cmsg_len الخاص بـ cmsghdr). بالنسبة لاستقبال الأخطاء، يتم تعيين علامة MSG_ERRQUEUE في msghdr. بعد تمرير خطأ، يتم إعادة إنشاء الخطأ الحالي للمقبس بناءً على الخطأ التالي المخزن وسيتم تمريره عند عملية المقبس التالية.
MSG_OOB
تطلب هذه العلامة استقبال بيانات خارج النطاق والتي لن يتم استقبالها في تدفق البيانات العادي. بعض البروتوكولات تضع البيانات السريعة في بداية طابور البيانات العادي، وبالتالي لا يمكن استخدام هذه العلامة مع هذه البروتوكولات.
MSG_PEEK
تتسبب هذه العلامة في أن تقوم عملية الاستقبال بإرجاع البيانات من بداية طابور الاستقبال دون إزالة هذه البيانات من الطابور. وبالتالي، فإن استدعاء استقبال لاحق سيعيد نفس البيانات.
MSG_TRUNC (منذ Linux 2.2)
بالنسبة لمقابس raw (AF_PACKET)، وحزم الإنترنت (منذ Linux 2.4.27/2.6.8)، و netlink (منذ Linux 2.6.22)، وحزم UNIX وكذلك sequenced-packet (منذ Linux 3.4): إرجاع الحجم الفعلي للحزمة أو البيانات، حتى عندما كانت أطول من المخزن المؤقت المقدم.
للاستخدام مع مقابس تدفق الإنترنت، انظر tcp(7).
MSG_WAITALL (منذ Linux 2.2)
تطلب هذه العلامة أن تمنع العملية حتى يتم تلبية الطلب بالكامل. ومع ذلك، قد ترجع الاستدعاء بيانات أقل من المطلوب إذا تم اعتراض إشارة، أو حدث خطأ أو انقطاع، أو كانت البيانات التالية التي سيتم استقبالها من نوع مختلف عن النوع الذي تم إرجاعه. هذه العلامة ليس لها تأثير على مقابس البيانات.

recvfrom()

تضع recvfrom() الرسالة المستقبلة في المخزن المؤقت buf. يجب على المستدعي تحديد حجم المخزن المؤقت في size.

إذا لم يكن src_addr فارغًا (NULL)، وإذا كان البروتوكول الأساسي يوفر عنوان المصدر للرسالة، يتم وضع عنوان المصدر هذا في المخزن المؤقت المشار إليه بواسطة src_addr. في هذه الحالة، addrlen هو وسيطة قيمة-نتيجة. قبل الاستدعاء، يجب تهيئتها بحجم المخزن المؤقت المرتبط بـ src_addr. عند العودة، يتم تحديث addrlen ليشمل الحجم الفعلي لعنوان المصدر. يتم اقتصاص العنوان المرجع إذا كان المخزن المؤقت المقدم صغيرًا جدًا؛ في هذه الحالة، سيعيد addrlen قيمة أكبر مما تم تقديمه للاستدعاء.

recv(2)

recv — receive messages from a socket

NAME

recv — receive messages from a socket

SYNOPSIS

#include <sys/types.h>
#include <sys/socket.h>

ssize_t recv(int socket, void *buffer, size_t length, int flags);

DESCRIPTION

The recv() function shall extract up to length bytes of data from the socket named by the descriptor socket and store them in the buffer pointed to by buffer.

The socket must be of a suitable type to receive data.

If the socket is of type SOCK_STREAM, SOCK_SEQPACKET, or SOCK_RDM, then the data received shall be the next length bytes available. If the data available is less than length bytes, then the call shall return with the number of bytes received.

If the socket is of type SOCK_DGRAM or SOCK_RAW, then the data received shall be a single datagram of at most length bytes in length. If the datagram is longer than length bytes, the excess bytes shall be discarded. If the datagram is shorter than length bytes, the entire datagram is returned.

If no message is currently available on the socket, the recv() call shall block until a message arrives, unless the socket is in non-blocking mode (see fcntl(2)).

The flags argument is an integer in which a bitwise OR of zero or more of the following flags may be included:

MSG_PEEK
Peek at incoming messages. The data is copied into the buffer but is not consumed from the input queue. If the message is longer than length bytes, recv() returns -1 and sets errno to EMSGSIZE. Subsequent recv() calls will reveal the same message, in its entirety, each time.
MSG_WAITALL
Wait for the entire amount of data requested to be received. The recv() call will return only after length bytes of data have been received, or an error occurs. If the socket is SOCK_STREAM or SOCK_SEQPACKET, and the end-of-file is detected before length bytes have been received, recv() returns the number of bytes received. If the socket is SOCK_DGRAM or SOCK_RAW, MSG_WAITALL has no effect.
MSG_DONTWAIT
Same as setting the O_NONBLOCK flag (see fcntl(2)) for the socket file description, i.e. the call will not block.
MSG_TRUNC
If the message is longer than length bytes, return only the first length bytes. If MSG_TRUNC is specified, and the message was truncated, the caller can detect this: if the value returned is equal to length, the message may have been truncated. The actual message length can be retrieved by using recvmsg(3) with the MSG_TRUNC flag.
MSG_CMSG
Retrieve control information. See getsockopt(3).
MSG_ERRQUEUE
Request to receive messages from the error queue.

RETURN VALUE

On success, recv() returns the number of bytes received. On error, -1 is returned, and errno is set to indicate the error.

If the peer has performed an orderly shutdown, an end-of-file will be returned (0 bytes read).

ERRORS

EBADF
socket is not a valid file descriptor.
ECONNABORTED
A virtual circuit was terminated due to data integrity errors or network failure.
ECONNRESET
A connection was forcibly closed by a peer.
EINTR
The call was interrupted by some delivery of a signal.
EINVAL
socket is not a socket, or flags has an invalid value.
ENOTSOCK
socket is a file, not a socket.
ENOMEM
Out of memory.
EOVERFLOW
The message is too long to fit in the specified buffer and MSG_TRUNC was not specified.
ETIMEDOUT
Connection timed out during recv().
EWOULDBLOCK
socket is in non-blocking mode and there are no messages available.

CONFORMING TO

POSIX.1-2001, POSIX.1-2008.

STANDARDS

None.

HISTORY

The recv() function call first appeared in 4.3BSD.

NOTES

The recv() function is equivalent to recvfrom(2) with a NULL from address argument.

It is recommended that applications use recvmsg(3) for more flexibility and control over message reception.

EXAMPLES

This example shows how to receive data from a TCP socket. 


#include <sys/types.h>
#include <sys/socket.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>

int main() {
    int sock_fd;
    char buffer[1024];
    ssize_t num_bytes;

    // Assume sock_fd is a connected socket descriptor
    // For example, obtained from accept()

    num_bytes = recv(sock_fd, buffer, sizeof(buffer) - 1, 0);

    if (num_bytes == -1) {
        perror("recv");
        exit(EXIT_FAILURE);
    } else if (num_bytes == 0) {
        printf("Connection closed by peer.\n");
    } else {
        buffer[num_bytes] = '\0'; // Null-terminate the received data
        printf("Received %zd bytes: %s\n", num_bytes, buffer);
    }

    close(sock_fd);
    return 0;
}

recv(2)

recv — استقبال الرسائل من مقبس

الاسم

recv — استقبال الرسائل من مقبس

الملخص

#include <sys/types.h>
#include <sys/socket.h>

ssize_t recv(int socket, void *buffer, size_t length, int flags);

الوصف

تقوم الدالة recv() باستخراج ما يصل إلى length بايت من البيانات من المقبس المسمى بواسطة الواصف socket وتخزينها في المخزن المؤقت المشار إليه بواسطة buffer.

يجب أن يكون المقبس من نوع مناسب لاستقبال البيانات.

إذا كان المقبس من النوع SOCK_STREAM أو SOCK_SEQPACKET أو SOCK_RDM، فإن البيانات المستلمة ستكون الـ length بايت التالية المتاحة. إذا كانت البيانات المتاحة أقل من length بايت، فستعود الدالة مع عدد البايتات المستلمة.

إذا كان المقبس من النوع SOCK_DGRAM أو SOCK_RAW، فإن البيانات المستلمة ستكون رسالة واحدة بحجم أقصى يبلغ length بايت. إذا كانت الرسالة أطول من length بايت، فسيتم تجاهل البايتات الزائدة. إذا كانت الرسالة أقصر من length بايت، فسيتم إرجاع الرسالة بأكملها.

إذا لم تكن هناك رسالة متاحة حاليًا على المقبس، فسوف تنتظر استدعاء recv() حتى تصل رسالة، ما لم يكن المقبس في وضع غير مانع (راجع fcntl(2)).

الوسيط flags هو عدد صحيح يمكن أن يتضمن فيه OR بت الصفري أو أكثر من الأعلام التالية:

MSG_PEEK
الاطلاع على الرسائل الواردة. يتم نسخ البيانات إلى المخزن المؤقت ولكن لا يتم استهلاكها من قائمة الإدخال. إذا كانت الرسالة أطول من length بايت، فسترجع recv() -1 وتعيين errno إلى EMSGSIZE. استدعاءات recv() اللاحقة ستكشف عن نفس الرسالة، بكاملها، في كل مرة.
MSG_WAITALL
انتظر حتى يتم استلام كامل كمية البيانات المطلوبة. ستعود استدعاء recv() فقط بعد استلام length بايت من البيانات، أو حدوث خطأ. إذا كان المقبس SOCK_STREAM أو SOCK_SEQPACKET، وتم اكتشاف نهاية الملف قبل استلام length بايت، فسترجع recv() عدد البايتات المستلمة. إذا كان المقبس SOCK_DGRAM أو SOCK_RAW، فإن MSG_WAITALL ليس له تأثير.
MSG_DONTWAIT
نفس تعيين العلم O_NONBLOCK (راجع fcntl(2)) لوصف ملف المقبس، أي أن الاستدعاء لن يمنع.
MSG_TRUNC
إذا كانت الرسالة أطول من length بايت، فسيتم إرجاع أول length بايت فقط. إذا تم تحديد MSG_TRUNC، وتم اقتطاع الرسالة، يمكن للمستدعي اكتشاف ذلك: إذا كانت القيمة المرتجعة تساوي length، فقد تم اقتطاع الرسالة. يمكن استرداد طول الرسالة الفعلي باستخدام recvmsg(3) مع العلم MSG_TRUNC.
MSG_CMSG
استرداد معلومات التحكم. راجع getsockopt(3).
MSG_ERRQUEUE
طلب استقبال الرسائل من طابور الأخطاء.

قيمة الإرجاع

عند النجاح، ترجع recv() عدد البايتات المستلمة. عند حدوث خطأ، يتم إرجاع -1، وتعيين errno للإشارة إلى الخطأ.

إذا قام الطرف المقابل بإغلاق منظم، فسيتم إرجاع نهاية الملف (0 بايت مقروءة).

الأخطاء

EBADF
socket ليس واصف ملف صالح.
ECONNABORTED
تم إنهاء دائرة افتراضية بسبب أخطاء سلامة البيانات أو فشل الشبكة.
ECONNRESET
تم إغلاق الاتصال قسريًا بواسطة طرف مقابل.
EINTR
تم مقاطعة الاستدعاء بواسطة بعض تسليم الإشارة.
EINVAL
socket ليس مقبسًا، أو أن flags لها قيمة غير صالحة.
ENOTSOCK
socket هو ملف، وليس مقبسًا.
ENOMEM
نفاد الذاكرة.
EOVERFLOW
الرسالة طويلة جدًا بحيث لا تتسع في المخزن المؤقت المحدد ولم يتم تحديد MSG_TRUNC.
ETIMEDOUT
انتهت مهلة الاتصال أثناء recv().
EWOULDBLOCK
socket في وضع غير مانع ولا توجد رسائل متاحة.

التوافق مع

POSIX.1-2001, POSIX.1-2008.

المعايير

لا يوجد.

التاريخ

ظهر استدعاء الدالة recv() لأول مرة في 4.3BSD.

ملاحظات

الدالة recv() مكافئة للدالة recvfrom(2) مع وسيط عنوان من NULL.

يوصى بأن تستخدم التطبيقات recvmsg(3) لمزيد من المرونة والتحكم في استقبال الرسائل.

أمثلة

يوضح هذا المثال كيفية استقبال البيانات من مقبس TCP. 


#include <sys/types.h>
#include <sys/socket.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>

int main() {
    int sock_fd;
    char buffer[1024];
    ssize_t num_bytes;

    // افترض أن sock_fd هو واصف مقبس متصل
    // على سبيل المثال، تم الحصول عليه من accept()

    num_bytes = recv(sock_fd, buffer, sizeof(buffer) - 1, 0);

    if (num_bytes == -1) {
        perror("recv");
        exit(EXIT_FAILURE);
    } else if (num_bytes == 0) {
        printf("تم إغلاق الاتصال بواسطة الطرف المقابل.\n");
    } else {
        buffer[num_bytes] = '\0'; // إنهاء البيانات المستلمة بـ null
        printf("تم استلام %zd بايت: %s\n", num_bytes, buffer);
    }

    close(sock_fd);
    return 0;
}
