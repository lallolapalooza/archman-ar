radvd.conf(5)

الحزمة: radvd

التراخيص: custom

RADVD.CONF(5) RADVD.CONF(5)

الاسم

radvd.conf - ملف إعدادات خدمة الإعلان عن الموجهات radvd

الوصف

يصف هذا الملف المعلومات التي يتم تضمينها في إعلان الموجه (RA) لواجهة محددة.

يحتوي الملف على تعريف واجهة واحد أو أكثر بالشكل:

interface name {
	قائمة خيارات الواجهة المحددة
	قائمة تعريفات البادئة
	قائمة العملاء (عناوين IPv6) للإعلان لهم
	قائمة تعريفات المسار
	قائمة تعريفات RDNSS
	قائمة تعريفات DNSSL
	قائمة تعريفات ABRO
	قائمة تعريفات NAT64 pref64
	قائمة بادئات التجاهل التلقائي
	قائمة عناوين مصدر RA المقبولة
};

تم تفصيل جميع خيارات الواجهة المحددة الممكنة أدناه. يجب إنهاء كل خيار بنقطة فاصلة. الخيارات ليست حساسة لحالة الأحرف.

تعريفات البادئة بالشكل:

prefix prefix/length {
	قائمة خيارات البادئة المحددة
};

يمكن أن تكون البادئة بادئة الشبكة أو عنوان الواجهة. يجب استخدام عنوان الواجهة عند استخدام امتدادات Mobile IPv6.

البادئة الخاصة "::/64" مدعومة أيضًا على الأنظمة التي تطبق getifaddrs() (على الأنظمة الأخرى، يفشل تفعيل الإعدادات ويخرج radvd). عند التكوين، يختار radvd جميع البادئات غير المحلية للرابط المعينة للواجهة ويبدأ في الإعلان عنها (ما لم يتم تجاهلها باستخدام autoignoreprefixes). قد يكون هذا مناسبًا في سيناريوهات غير 6to4 حيث قد تتغير البادئة الصاعدة. هذا الخيار غير متوافق مع خيار Base6to4Interface. يتم تمكين خيار AdvRouterAddr دائمًا عند استخدام هذا الإعداد.

تم وصف جميع خيارات البادئة المحددة الممكنة أدناه. يجب إنهاء كل خيار بنقطة فاصلة.

يُسمح بالقيم العشرية فقط لـ MinDelayBetweenRAs و MaxRtrAdvInterval و MinRtrAdvInterval. يجب استخدام القيم العشرية فقط عند استخدام امتدادات Mobile IPv6.

تعريفات المسار بالشكل:

route prefix/length {
	قائمة خيارات المسار المحددة
};

يجب أن تكون بادئة تعريف المسار بادئة شبكة؛ يمكن استخدامها للإعلان عن مسارات أكثر تحديدًا للمضيفين.

تعريفات RDNSS (خادم DNS العودي) بالشكل:

RDNSS ip [ip] [ip] {
	قائمة خيارات rdnss المحددة
};

يحتوي كل كتلة تعريف RDNSS على حد أقصى للطول يبلغ 2040 بايت بما في ذلك الرأس، وهذا يستوعب ما يصل إلى 127 عنوانًا.

إذا تم تجاوز الطول، سيقوم radvd بتسجيل خطأ غير قاتل بدلاً من إرسال الخيار.

تعريفات DNSSL (قائمة بحث DNS) بالشكل:

DNSSL suffix [suffix] [suffix] [...] {
	قائمة خيارات dnssl المحددة
};

يحتوي كل كتلة تعريف DNSSL على حد أقصى للطول يبلغ 2040 بايت بما في ذلك الرأس، ويتم تطبيقه بعد ترميز اللواحق وفقًا لـ RFC1035، القسم 3.1.

إذا تم تجاوز الطول، سيقوم radvd بتسجيل خطأ غير قاتل بدلاً من إرسال الخيار.

بشكل افتراضي، سيرسل radvd إعلانات الموجهات المتعددة بحيث يمكن لكل عقدة على الرابط استخدامها. يمكن تكوين قائمة العملاء (عنوان IPv6) للإعلان لهم، واستقبال طلبات المسار منهم. إذا تم ذلك، فلن يرسل radvd رسائل إلى عناوين البث المتعدد ولكن إلى العناوين الأحادية المكونة فقط. يتم رفض الطلبات من عناوين أخرى ما لم يتم تمكين UnrestrictedUnicast. هذا مشابه لـ UnicastOnly ولكنه يشمل الرسائل الدورية وتكوين وصول العميل الوارد. انظر قسم الأمثلة لحالة استخدام هذا.

التعريفات بالشكل:

clients {
        قائمة عناوين IPv6
};

يمكن أن تبدأ العملاء بـ "!" لتجاهلها بالكامل وعدم إرسال إعلانات لهم أبدًا.

بشكل افتراضي، سيستخدم radvd أول عنوان محلي للرابط للواجهة كعنوان مصدر لإعلانات المسار. يمكن تجاوز ذلك عن طريق تعيين قائمة عناوين المصدر المقبولة يدويًا. إذا تم ذلك، فسيستخدم radvd العنوان الأول من الواجهة الموجود في العناوين المصدر المكونة فقط. لن تقوم هذه الوظيفة بتزوير عنوان المصدر، ولكنها قد تكون مفيدة بالاشتراك مع VRRP أو وظائف أخرى تقوم بـ MAC افتراضي / تجاوز الفشل.

AdvRASrcAddress {
        قائمة عناوين IPv6
};

تعريفات ABRO (خيار الموجه الحدودي الموثوق به) بالشكل:

abro IPv6-address {
        قائمة خيارات abro المحددة
};

NAT64 pref64 (بادئة NAT64 المدرجة في إعلانات الموجه):

nat64prefix prefix/length {
	قائمة خيارات بادئة NAT64 المحددة
};

يمكن أن يكون قيمة length فقط /32 أو /40 أو /48 أو /56 أو /64 أو /96.

عند استخدام البادئة الخاصة "::/64"، تشكل هذه الخيار قائمة تجاهل للبادئات التي لا ينبغي إنشاؤها والإعلان عنها تلقائيًا. ليس له تأثير على أي تعريف بادئة آخر.

التعريفات بالشكل:

autoignoreprefixes {
        قائمة بادئات IPv6
};

خيارات الواجهة المحددة

IgnoreIfMissing on|off

علامة تشير إلى ما إذا كانت الواجهة يتم تجاهلها أم لا إذا كانت غير موجودة عند بدء التشغيل.

هذا مفيد للواجهات الديناميكية التي لا تكون نشطة عند بدء تشغيل radvd أو التي يتم تعطيلها وإعادة تمكينها ديناميكيًا أثناء تشغيل radvd.

الإصدارات الحالية من radvd تحاول تلقائيًا إعادة تمكين الواجهات.

تمكين IgnoreIfMissing يؤدي أيضًا إلى كتم بعض التحذيرات في رسائل السجل المتعلقة بالواجهات المفقودة.

الافتراضي: on

AdvSendAdvert on|off

علامة تشير إلى ما إذا كان الموجه يرسل إعلانات الموجه الدورية ويستجيب لطلبات الموجه.

لم يعد هذا الخيار بحاجة إلى تحديده أولاً، ولكنه يحتاج إلى أن يكون on لتمكين الإعلان على هذه الواجهة.

الافتراضي: off

UnicastOnly on|off

يشير إلى أن نوع رابط الواجهة يدعم البث الأحادي فقط. سيمنع هذا الإعلانات غير المطلوبة من الإرسال، وسيؤدي إلى إرسال الإعلانات المطلوبة بشكل أحادي إلى العقدة التي طلبتها. هذا الخيار ضروري للروابط المتعددة التي لا تدعم البث، مثل ISATAP.

الافتراضي: off

UnrestrictedUnicast on|off

علامة تشير إلى ما إذا كان سيتم الاستجابة لطلبات الموجه عندما تكون هناك قائمة بالعملاء المكونة.

يسمح هذا بإرسال الإعلانات غير المطلوبة بانتظام إلى بعض العملاء دون تجاهل طلبات الموجه من العملاء غير المعروفين.

الافتراضي: off

AdvRASolicitedUnicast on|off

يشير إلى أنه سيتم الاستجابة لطلبات الموجه بإعلانات موجه أحادية، كما هو موصى به في RFC7772. قد تواجه الشبكات الكبيرة ذات التركيز العالي للأجهزة المحمولة استنزافًا للبطارية، عندما يتم إرسال رسائل إعلان الموجه المطلوبة بشكل متعدد.

هذا يتوافق مع خيار Cisco IOS ipv6 nd ra solicited unicast

الافتراضي: on

MaxRtrAdvInterval seconds

الحد الأقصى للوقت المسموح به بين إرسال إعلانات الموجه المتعددة غير المطلوبة من الواجهة، بالثواني.

يجب ألا تقل عن 4 ثوانٍ ولا تزيد عن 1800 ثانية.

الحد الأدنى عند استخدام امتدادات Mobile IPv6: 0.07.

بالنسبة للقيم التي تقل عن 0.2 ثانية، يتم إضافة 0.02 ثانية لحساب دقة الجدولة كما هو محدد في RFC3775.

الافتراضي: 600 ثانية

MinRtrAdvInterval seconds

الحد الأدنى للوقت المسموح به بين إرسال إعلانات الموجه المتعددة غير المطلوبة من الواجهة، بالثواني.

يجب ألا تقل عن 3 ثوانٍ ولا تزيد عن 0.75 * MaxRtrAdvInterval.

هذا الموقع غير مرتبط بـ Arch Linux ولا تابع له بأي شكل من الأشكال. المحتوى هو ترجمة فقط لصفحات دليل Arch المنشورة على man.archlinux.org.

>الحد الأدنى عند استخدام ملحقات Mobile IPv6: 0.03.

الافتراضي: 0.33 * MaxRtrAdvInterval

MinDelayBetweenRAs ثواني

الحد الأدنى للوقت المسموح به بين إرسال إعلانات الموجه المجمعة من الواجهة، بالثواني.

ينطبق هذا على إعلانات الموجه المجمعة المطلوبة. يُعرّف هذا كثابت البروتوكول MIN_DELAY_BETWEEN_RAS في RFC4861. يعيد MIPv6 تعريف هذه المعلمة ليكون لها حد أدنى قدره 0.03 ثانية.

الحد الأدنى عند استخدام ملحقات Mobile IPv6: 0.03.

الافتراضي: 3

AdvManagedFlag on|off

عند ضبطه، تستخدم المضيفات البروتوكول المُدار (الحالة) للتكوين التلقائي للعناوين بالإضافة إلى أي عناوين تم تكوينها تلقائيًا باستخدام التكوين التلقائي للعناوين عديم الحالة. يتم وصف استخدام هذه العلامة في RFC 4862.

الافتراضي: off

AdvOtherConfigFlag on|off

عند ضبطه، تستخدم المضيفات البروتوكول المُدار (الحالة) للتكوين التلقائي للمعلومات الأخرى (غير العناوين). يتم وصف استخدام هذه العلامة في RFC 4862.

الافتراضي: off

AdvLinkMTU عدد صحيح

يُستخدم خيار MTU في رسائل إعلان الموجه لضمان أن جميع العقد على الرابط تستخدم نفس قيمة MTU في الحالات التي لا تكون فيها MTU الرابط معروفة جيدًا.

إذا تم تحديده، أي إذا لم يكن 0، يجب ألا يكون أصغر من 1280 ولا أكبر من الحد الأقصى لـ MTU المسموح به لهذا الرابط (على سبيل المثال، لدى Ethernet حد أقصى لـ MTU يبلغ 1500. انظر RFC 4864).

الافتراضي: 0

AdvReachableTime مللي ثانية

الوقت، بالمللي ثانية، الذي تفترض فيه العقدة أن الجار قابل للوصول إليه بعد تلقي تأكيد الوصول. يستخدم بواسطة خوارزمية الكشف عن عدم وصول الجيران (انظر القسم 7.3 من RFC 4861). القيمة صفر تعني غير محددة (بواسطة هذا الموجه).

يجب ألا تكون أكبر من 3,600,000 مللي ثانية (1 ساعة).

الافتراضي: 0

AdvRetransTimer مللي ثانية

الوقت، بالمللي ثانية، بين رسائل طلب الجار المعاد إرسالها. يستخدم بواسطة تحديد العناوين وخوارزمية الكشف عن عدم وصول الجيران (انظر القسمين 7.2 و 7.3 من RFC 4861). القيمة صفر تعني غير محددة (بواسطة هذا الموجه).

الافتراضي: 0

AdvCurHopLimit عدد صحيح

القيمة الافتراضية التي يجب وضعها في حقل عدد القفزات لرأس IP لحزم IP الصادرة (الأحادية). يجب تعيين القيمة على القطر الحالي للإنترنت. القيمة صفر تعني غير محددة (بواسطة هذا الموجه).

الافتراضي: 64

AdvDefaultLifetime ثواني

العمر الافتراضي المرتبط بالموجه الافتراضي بوحدات الثواني. الحد الأقصى للقيمة يقابل 18.2 ساعة. العمر الافتراضي 0 يشير إلى أن الموجه ليس موجهًا افتراضيًا ولا يجب أن يظهر في قائمة الموجهات الافتراضية. يطبق العمر الافتراضي للموجه فقط على فائدة الموجه كموجه افتراضي؛ لا ينطبق على المعلومات الواردة في حقول الرسائل أو الخيارات الأخرى. الخيارات التي تحتاج إلى حدود زمنية لمعلوماتها تتضمن حقول العمر الافتراضي الخاصة بها.

يجب أن تكون إما صفرًا أو بين MaxRtrAdvInterval و 9000 ثانية.

الافتراضي: 3 * MaxRtrAdvInterval (الحد الأدنى 1 ثانية).

AdvDefaultPreference low|medium|high

التفضيل المرتبط بالموجه الافتراضي، إما "منخفض" أو "متوسط" أو "مرتفع".

الافتراضي: medium

AdvSourceLLAddress on|off

عند ضبطه، يتم تضمين عنوان طبقة الارتباط للواجهة الصادرة في RA.

الافتراضي: on

RemoveAdvOnExit on|off

عند الإغلاق، أرسل إعلانًا نهائيًا بعمر افتراضي صفر للموجه. يجب أن يتسبب هذا في إزالة الموجه والمسارات على الفور من جدول المسارات الخاص بعقد النهاية المستقبلة. قد تحتاج هذه الميزة إلى تعطيلها ("off") في إعداد vrrp أو carp.

الافتراضي: on

AdvHomeAgentFlag on|off

عند ضبطه، يشير إلى أن الموجه المرسل قادر على العمل كوكيل Home Agent لـ Mobile IPv6. عند ضبطه، يتم استخدام الحدود الدنيا المحددة بواسطة Mobile IPv6 لـ MinRtrAdvInterval و MaxRtrAdvInterval.

الافتراضي: off

AdvHomeAgentInfo on|off

عند ضبطه، يتم تضمين خيار معلومات وكيل Home Agent (المحدد بواسطة Mobile IPv6) في إعلانات الموجه. يجب أيضًا تعيين AdvHomeAgentFlag عند استخدام هذا الخيار.

الافتراضي: off

HomeAgentLifetime ثواني

مدة الوقت بالثواني (بالنسبة لوقت إرسال الحزمة) التي يقدم فيها الموجه خدمات وكيل Home Agent لـ Mobile IPv6. يجب ألا يتم استخدام القيمة 0. الحد الأقصى للعمر الافتراضي هو 65520 ثانية (18.2 ساعة). يتم تجاهل هذا الخيار إذا لم يتم تعيين AdvHomeAgentInfo.

إذا تم تعيين كل من HomeAgentLifetime و HomeAgentPreference على قيمهما الافتراضية، فلن يتم إرسال خيار معلومات وكيل Home Agent.

الافتراضي: AdvDefaultLifetime

HomeAgentPreference عدد صحيح

تفضيل وكيل Home Agent الذي يرسل إعلان الموجه هذا. تشير القيم التي تزيد عن 0 إلى وكيل Home Agent أكثر تفضيلاً، وتشير القيم الأقل من 0 إلى وكيل Home Agent أقل تفضيلاً. يتم تجاهل هذا الخيار إذا لم يتم تعيين AdvHomeAgentInfo.

إذا تم تعيين كل من HomeAgentLifetime و HomeAgentPreference على قيمهما الافتراضية، فلن يتم إرسال خيار معلومات وكيل Home Agent.

الافتراضي: 0

AdvMobRtrSupportFlag on|off

عند ضبطه، يشير وكيل Home Agent إلى أنه يدعم تسجيلات الموجه المتنقل (المحددة بواسطة NEMO Basic). يجب أيضًا تعيين AdvHomeAgentInfo عند استخدام هذا الخيار.

الافتراضي: off

AdvIntervalOpt on|off

عند ضبطه، يتم تضمين خيار فاصل الإعلان (المحدد بواسطة Mobile IPv6) في إعلانات الموجه. عند ضبطه، يتم استخدام الحدود الدنيا المحددة بواسطة Mobile IPv6 لـ MinRtrAdvInterval و MaxRtrAdvInterval.

يعتمد فاصل الإعلان على معلمة MaxRtrAdvInterval المكونة، ما لم تكن هذه أقل من 200 مللي ثانية. في هذه الحالة، يكون الفاصل المعلن هو (MaxRtrAdvInterval + 20 مللي ثانية).

الافتراضي: off

AdvCaptivePortalAPI "URL"

عند ضبطه، قم بالإعلان عن عنوان URL لواجهة برمجة تطبيقات بوابة احتجاز RFC8908.

راجع RFC8952 بنية البوابة الاحتجازية، RFC8910 تحديد البوابة الاحتجازية في DHCP وإعلانات الموجه (RAs)، و RFC8908 واجهة برمجة تطبيقات البوابة الاحتجازية لمزيد من المعلومات.

على الأرجح أنك لا تحتاج إلى هذا.

الافتراضي: غير مضمن

خيارات محددة لـ PREFIX

AdvOnLink on|off

عند ضبطه، يشير إلى أن هذا البادئة يمكن استخدامها لتحديد على الرابط. عندما لا يتم ضبطه، لا يقدم الإعلان أي بيان حول خصائص البادئة على الرابط أو خارجه. على سبيل المثال، قد يتم استخدام البادئة لتكوين عناوين مع بعض العناوين التي تنتمي إلى البادئة كونها على الرابط والبعض الآخر خارج الرابط.

الافتراضي: on

AdvAutonomous on|off

عند ضبطه، يشير إلى أن هذه البادئة يمكن استخدامها للتكوين التلقائي للعناوين كما هو محدد في RFC 4862.

الافتراضي: on

AdvRouterAddr on|off

عند ضبطه، يشير إلى أن عنوان الواجهة يتم إرساله بدلاً من بادئة الشبكة، كما هو مطلوب بواسطة Mobile IPv6. عند ضبطه، يتم استخدام الحدود الدنيا المحددة بواسطة Mobile IPv6 لـ MinRtrAdvInterval و MaxRtrAdvInterval.

الافتراضي: off

AdvValidLifetime ثواني|infinity

مدة الوقت بالثواني (بالنسبة لوقت إرسال الحزمة) التي تكون فيها البادئة صالحة لغرض تحديد على الرابط. القيمة الرمزية infinity تمثل اللانهاية (أي قيمة كل البتات واحد (0xffffffff)). العمر الافتراضي الصالح يستخدم أيضًا بواسطة RFC 4862.

لاحظ أن العملاء سيتجاهلون AdvValidLifetime لبادئة موجودة إذا كان العمر الافتراضي أقل من ساعتين، كما هو مطلوب في RFC 4862 القسم 5.5.3 النقطة e).

ملاحظة: القيمة الافتراضية المقترحة لـ RFC4861 أطول بكثير: 30 يومًا.

الافتراضي: 86400 ثانية (1 يوم)

AdvPreferredLifetime ثواني|infinity

DeprecatePrefix on|off

عند الإغلاق، سيؤدي هذا الخيار إلى قيام radvd بإلغاء صلاحية البادئة عن طريق الإعلان عنها في إعلان جهاز التوجيه (RA) الخاص بإغلاق radvd بعمر تفضيلي صفر وعمر صالح أكبر بقليل من ساعتين. سيشجع هذا العقد النهائية التي تستخدم هذه البادئة على إلغاء صلاحية أي عناوين مرتبطة بها على الفور. لاحظ أنه يجب استخدام هذا الخيار فقط عندما يعلن جهاز توجيه واحد فقط عن البادئة على الرابط، وإلا فإن العقد النهائية ستبطل العناوين المرتبطة بها على الرغم من أن البادئة لا تزال صالحة للاستخدام المفضل.

انظر RFC4862، القسم 5.5.3، "معالجة إعلانات جهاز التوجيه" (Router Advertisement Processing)، الجزء (e).

الافتراضي: off

DecrementLifetimes on|off

يؤدي هذا الخيار إلى قيام radvd بتقليل قيم الأعمار التفضيلية والصالحة للبادئة بمرور الوقت. يتم تقليل الأعمار بمقدار الثواني منذ آخر RA. إذا تلقى radvd إشارة SIGUSR1، فسيقوم بإعادة تعيين قيم الأعمار التفضيلية والصالحة إلى القيم الأولية التي استخدمها radvd عند بدء تشغيله. إذا لم يتلق radvd إشارة SIGUSR1 أبدًا، فسيستمر في تقليل الأعمار حتى يصل العمر المفضل إلى الصفر. بعد إعلان RA نهائي بعمر تفضيلي بقيمة صفر، سيتوقف radvd عن الإعلان عن البادئة. إذا أدت إشارة SIGUSR1 بعد ذلك إلى إعادة تعيين الأعمار، فستظهر البادئة مرة أخرى في إعلانات RA.

يهدف هذا الخيار إلى استخدامه بالاقتران مع عميل DHCPv6 الذي يستخدم خيار Association for Prefix Delegation (IA_PD) لاكتساب بادئة من جهاز توجيه تفويضي (Delegating Router) لاستخدامها من قبل جهاز توجيه طالب (Requesting Router). في هذا السيناريو، ستتقادم البادئات من داخل البادئة المفوضة التي يعلن عنها radvd بالتوازي وبنفس المعدل مثل البادئة المفوضة، وتنتهي صلاحيتها في نفس الوقت تقريبًا، إذا لم يتم تمديد عمر البادئة المفوضة.

انظر RFC3633، "خيارات بادئة IPv6 لبروتوكول التكوين الديناميكي للمضيف (DHCP) الإصدار 6" (IPv6 Prefix Options for Dynamic Host Configuration Protocol (DHCP) version 6).

الافتراضي: off

Base6Interface name

إذا تم تحديد هذا الخيار، فسيتم دمج هذه البادئة مع عنوان IPv6 للوجهة المحددة بواسطة name. سيكون طول البادئة الناتجة 64.

Base6to4Interface name

إذا تم تحديد هذا الخيار، فسيتم دمج هذه البادئة مع عنوان IPv4 للوجهة name لإنتاج بادئة 6to4 صالحة. سيتم استبدال أول 16 بت من هذه البادئة بـ 2002 وسيتم استبدال الـ 32 بت التالية من هذه البادئة بعنوان IPv4 المعين للوجهة name في وقت التكوين. سيتم الإعلان عن الـ 80 بت المتبقية من البادئة (بما في ذلك معرف SLA ID) كما هو محدد في ملف التكوين. انظر القسم التالي للحصول على مثال.

إذا لم تكن الوجهة name متاحة في وقت التكوين، فسيتم كتابة تحذير في السجل وسيتم تعطيل هذه البادئة حتى يتم إعادة تكوين radvd.

يمكّن هذا الخيار الأنظمة التي لديها عناوين IPv4 ديناميكية من تحديث بادئات 6to4 المعلن عنها ببساطة عن طريق إعادة تشغيل radvd أو إرسال إشارة SIGHUP لجعل radvd يعيد تكوين نفسه.

لاحظ أنه يجب الإعلان عن بادئات 6to4 المشتقة من عناوين IPv4 المعينة ديناميكيًا بعمر بادئة أقصر بكثير (انظر الخيارات AdvValidLifetime و AdvPreferredLifetime).

لمزيد من المعلومات حول 6to4، انظر RFC 3056.

الافتراضي: لا يتم استخدام 6to4

أمثلة

interface eth0
{
        AdvSendAdvert on;
        prefix 2001:db8:0:1::/64
        {
                AdvOnLink on;
                AdvAutonomous on;
        };
};

يقول أن خادم الإعلان عن أجهزة التوجيه يجب أن يعلن (AdvSendAdvert on;) عن البادئة 2001:db8:0:1:: التي لها طول 64 على الواجهة eth0. أيضًا، يجب تمييز البادئة كذاتية (AdvAutonomous on;) وكـ على الرابط (AdvOnLink on;). تترك جميع الخيارات الأخرى بقيمها الافتراضية.

لدعم اكتشاف الحركة لعقد Mobile IPv6 المتنقلة، يجب استخدام عنوان الواجهة بدلاً من بادئة الشبكة:

interface eth0
{
        AdvSendAdvert on;
        prefix 2001:db8:0:1::4/64
        {
                AdvOnLink on;
                AdvAutonomous on;
                AdvRouterAddr on;
        };
};

لدعم 6to4، قم بتضمين الخيار Base6to4Interface في كل قسم بادئة. عند استخدام عنوان IPv4 ديناميكي، اضبط أعمار بادئات قصيرة لمنع المضيفين من الاحتفاظ بالبادئات غير القابلة للوصول بعد تعيين عنوان IPv4 جديد. عند الإعلان على واجهة ديناميكية (على سبيل المثال، Bluetooth)، تخطى الواجهة إذا لم تكن نشطة بعد.

interface bnep0
{
        IgnoreIfMissing on;
        AdvSendAdvert on;
        # الإعلان كل 30 ثانية على الأقل
        MaxRtrAdvInterval 30;
        prefix 0:0:0:5678::/64
        {
                AdvOnLink on;
                AdvAutonomous on;
                Base6to4Interface ppp0;
                # أعمار قصيرة جدًا للعناوين الديناميكية
                AdvValidLifetime 300;
                AdvPreferredLifetime 120;
        };
};

نظرًا لأن 6to4 ممكّن، سيتم الإعلان عن البادئة كـ 2002:WWXX:YYZZ:5678::/64، حيث WW.XX.YY.ZZ هو عنوان IPv4 الخاص بـ ppp0 في وقت التكوين. (تُكتب عناوين IPv6 بالترميز الست عشري بينما تُكتب عناوين IPv4 بالأرقام العشرية، لذا سيتم تمثيل عنوان IPv4 WW.XX.YY.ZZ في بادئة 6to4 بالترميز الست عشري.)

في هذه الحالة المحددة، قد ترسل نصوص البرمجة النصية للتكوين إشارة HUP إلى radvd عند تشغيل bnep0
