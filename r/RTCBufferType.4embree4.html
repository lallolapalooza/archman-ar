#include <embree4/rtcore_ray.h>
enum RTCBufferType
{
  RTC_BUFFER_TYPE_INDEX            = 0,
  RTC_BUFFER_TYPE_VERTEX           = 1,
  RTC_BUFFER_TYPE_VERTEX_ATTRIBUTE = 2,
  RTC_BUFFER_TYPE_NORMAL           = 3,
  RTC_BUFFER_TYPE_TANGENT          = 4,
  RTC_BUFFER_TYPE_NORMAL_DERIVATIVE = 5,
  RTC_BUFFER_TYPE_GRID                 = 8,
  RTC_BUFFER_TYPE_FACE                 = 16,
  RTC_BUFFER_TYPE_LEVEL                = 17,
  RTC_BUFFER_TYPE_EDGE_CREASE_INDEX    = 18,
  RTC_BUFFER_TYPE_EDGE_CREASE_WEIGHT   = 19,

الوصف

يحدد هيكل RTBufferType خانات لتخصيص المخازن المؤقتة للبيانات باستخدام استدعاءات واجهة برمجة التطبيقات [rtcSetGeometryBuffer]، [rtcSetSharedGeometryBuffer]، و [rtcSetNewGeometryBuffer].

بالنسبة لمعظم أنواع الهندسة، يتم استخدام خانة RTC_BUFFER_TYPE_INDEX لتخصيص مخزن مؤقت للفهرس، بينما يتم استخدام خانة RTC_BUFFER_TYPE_VERTEX لتخصيص المخزن المؤقت للرؤوس المقابل.

يمكن استخدام خانة RTC_BUFFER_TYPE_VERTEX_ATTRIBUTE لتخصيص بيانات إضافية للأشكال الهندسية يمكن استيفاؤها باستخدام استدعاء واجهة برمجة التطبيقات [rtcInterpolate].

تعد خانات RTC_BUFFER_TYPE_NORMAL و RTC_BUFFER_TYPE_TANGENT و RTC_BUFFER_TYPE_NORMAL_DERIVATIVE مخازن مؤقتة خاصة مطلوبة لتخصيص المتجهات العمودية والزوايا المشتقة لكل رأس لبعض أنواع المنحنيات.

يتم استخدام مخزن RTC_BUFFER_TYPE_GRID المؤقت لتخصيص مخزن مؤقت لأشكال الشبكة الهندسية (انظر [RTC_GEOMETRY_TYPE_GRID]).

تعد خانات RTC_BUFFER_TYPE_FACE و RTC_BUFFER_TYPE_LEVEL و RTC_BUFFER_TYPE_EDGE_CREASE_INDEX و RTC_BUFFER_TYPE_EDGE_CREASE_WEIGHT و RTC_BUFFER_TYPE_VERTEX_CREASE_INDEX و RTC_BUFFER_TYPE_VERTEX_CREASE_WEIGHT و RTC_BUFFER_TYPE_HOLE مخازن مؤقتة خاصة مطلوبة لإنشاء شبكات التقسيم (انظر [RTC_GEOMETRY_TYPE_SUBDIVISION]).

يتم استخدام مخزن RTC_BUFFER_TYPE_TRANSFORM المؤقت لتوفير معلومات التحويل الفرعية لأشكال مصفوفة المثيلات (انظر [RTC_GEOMETRY_TYPE_INSTANCE_ARRAY]).

يمكن استخدام خانة RTC_BUFFER_TYPE_FLAGS لإضافة علامات إضافية لكل شكل فرعي هندسي، وهي تستخدم حاليًا فقط للمنحنيات الخطية.
