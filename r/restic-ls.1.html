restic-ls(1)

الحزمة: restic

التراخيص: BSD

restic backup(1) restic backup(1)

الاسم

restic-ls - سرد الملفات في لقطة

الملخص

restic ls [علامات] معرف_اللقطة [دليل...]

الوصف

الأمر "ls" يسرد الملفات والأدلة في لقطة.

يمكن استخدام معرف اللقطة الخاص "latest" لسرد الملفات والأدلة في أحدث لقطة في المستودع. يمكن استخدام العلامة --host بالاشتراك معها لتحديد أحدث لقطة قادمة من مضيف معين فقط.

يمكن تصفية قوائم الملفات اختياريًا حسب الأدلة. يتم تفسير أي حجج موضعية بعد معرف اللقطة كمسارات أدلة مطلقة، وسيتم سرد الملفات الموجودة داخل تلك الأدلة فقط. إذا تم استخدام العلامة --recursive، فسيسمح التصفية بالمرور إلى المجلدات الفرعية للأدلة المتطابقة. يجب أن تكون مسارات الأدلة المحددة مطلقة (تبدأ بفاصل مسار)؛ تستخدم المسارات الشرطة المائلة للأمام '/' كفاصل.

يمكن فرز قوائم الملفات عن طريق تحديد --sort متبوعة بأحد محددات الفرز (name|size|time=mtime|atime|ctime|extension). يمكن عكس الفرز عن طريق تحديد --reverse.

حالة الخروج

حالة الخروج هي 0 إذا كان الأمر ناجحًا. حالة الخروج هي 1 إذا كان هناك أي خطأ. حالة الخروج هي 10 إذا كان المستودع غير موجود. حالة الخروج هي 11 إذا كان المستودع مقفلاً بالفعل. حالة الخروج هي 12 إذا كانت كلمة المرور غير صحيحة.

خيارات

-h, --help[=false] المساعدة لـ ls

-H, --host=[] خذ في الاعتبار اللقطات لهذا المضيف فقط، عند إعطاء معرف اللقطة "latest" (يمكن تحديده عدة مرات) (الافتراضي: $RESTIC_HOST)

--human-readable[=false] طباعة الأحجام بتنسيق سهل القراءة

-l, --long[=false] استخدام تنسيق قائمة طويل يعرض الحجم والوضع

--ncdu[=false] إخراج تنسيق تصدير NCDU (توجيه إلى 'ncdu -f -')

--path=[] خذ في الاعتبار اللقطات التي تتضمن هذا المسار (المطلق) فقط، عند إعطاء معرف اللقطة "latest" (يمكن تحديده عدة مرات، يجب أن تتضمن اللقطات جميع المسارات المحددة)

--recursive[=false] تضمين الملفات في المجلدات الفرعية للأدلة المدرجة

--reverse[=false] عكس الإخراج المصنف

-s, --sort=name فرز الإخراج حسب (name|size|time=mtime|atime|ctime|extension)

--tag=[] خذ في الاعتبار اللقطات التي تتضمن tag[,tag,...] فقط، عند إعطاء معرف اللقطة "latest" (يمكن تحديده عدة مرات)

الخيارات الموروثة من الأوامر الأصل

--cacert=[] ملف لتحميل شهادات الجذر منه (الافتراضي: استخدام شهادات النظام أو $RESTIC_CACERT)

--cache-dir="" تعيين دليل ذاكرة التخزين المؤقت. (الافتراضي: استخدام دليل ذاكرة التخزين المؤقت الافتراضي للنظام)

--cleanup-cache[=false] إزالة تلقائية لأدلة ذاكرة التخزين المؤقت القديمة

--compression=auto وضع الضغط (متوفر فقط لإصدار تنسيق المستودع 2)، أحد (auto|off|max) (الافتراضي: $RESTIC_COMPRESSION)

--http-user-agent="" تعيين وكيل مستخدم http لطلبات http الصادرة

--insecure-no-password[=false] استخدام كلمة مرور فارغة للمستودع، يجب تمريرها إلى كل أمر restic (غير آمن)

--insecure-tls[=false] تخطي التحقق من شهادة TLS عند الاتصال بالمستودع (غير آمن)

--json[=false] تعيين وضع الإخراج إلى JSON للأوامر التي تدعمه

--key-hint="" معرف المفتاح للمفتاح لمحاولة فك التشفير أولاً (الافتراضي: $RESTIC_KEY_HINT)

--limit-download=0 يحد التنزيلات إلى معدل أقصى بالكيلوبايت/ثانية. (الافتراضي: غير محدود)

--limit-upload=0 يحد التحميلات إلى معدل أقصى بالكيلوبايت/ثانية. (الافتراضي: غير محدود)

--no-cache[=false] عدم استخدام ذاكرة تخزين مؤقت محلية

--no-extra-verify[=false] تخطي التحقق الإضافي من البيانات قبل التحميل (انظر التوثيق)

--no-lock[=false] عدم قفل المستودع، هذا يسمح ببعض العمليات على المستودعات للقراءة فقط

-o, --option=[] تعيين خيار موسع (key=value، يمكن تحديده عدة مرات)

--pack-size=0 تعيين حجم العبوة المستهدف بالميبي بايت، قد تكون ملفات العبوة المنشأة أكبر (الافتراضي: $RESTIC_PACK_SIZE)

--password-command="" أمر shell للحصول على كلمة مرور المستودع منه (الافتراضي: $RESTIC_PASSWORD_COMMAND)

-p, --password-file="" ملف لقراءة كلمة مرور المستودع منه (الافتراضي: $RESTIC_PASSWORD_FILE)

-q, --quiet[=false] عدم إخراج تقرير تقدم شامل

-r, --repo="" المستودع للنسخ الاحتياطي إليه أو الاستعادة منه (الافتراضي: $RESTIC_REPOSITORY)

--repository-file="" ملف لقراءة موقع المستودع منه (الافتراضي: $RESTIC_REPOSITORY_FILE)

--retry-lock=0s إعادة المحاولة لقفل المستودع إذا كان مقفلاً بالفعل، يأخذ قيمة مثل 5m أو 2h (الافتراضي: لا توجد محاولات إعادة)

--stuck-request-timeout=5m0s مدة بعد إعادة المحاولة للطلبات العالقة

--tls-client-cert="" مسار إلى ملف يحتوي على شهادة عميل TLS بتنسيق PEM ومفتاح خاص (الافتراضي: $RESTIC_TLS_CLIENT_CERT)

-v, --verbose[=0] كن مطولاً (حدد عدة مرات أو مستوى باستخدام --verbose=n``، الحد الأقصى للمستوى/المرات هو 2)
