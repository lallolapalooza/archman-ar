regex.h(0P) دليل مبرمج POSIX regex.h(0P)

مقدمة

هذه الصفحة الدليلية هي جزء من دليل مبرمج POSIX. قد يختلف تنفيذ Linux لهذا الواجهة (راجع صفحة دليل Linux المقابلة لتفاصيل سلوك Linux)، أو قد لا تكون الواجهة مطبقة على Linux.

الاسم

regex.h — أنواع مطابقة التعبيرات العادية

الملخص

#include <regex.h>

الوصف

يقوم الرأس <regex.h> بتعريف الهياكل والثوابت الرمزية المستخدمة في دوال regcomp() و regexec() و regerror() و regfree().

يقوم الرأس <regex.h> بتعريف نوع الهيكل regex_t، والذي يجب أن يتضمن على الأقل العضو التالي:

size_t    re_nsub    عدد الأقواس الفرعية.

يقوم الرأس <regex.h> بتعريف النوع size_t كما هو موضح في <sys/types.h>.

يقوم الرأس <regex.h> بتعريف النوع regoff_t كنوع عدد صحيح موقع يمكنه استيعاب أكبر قيمة يمكن تخزينها في نوع ptrdiff_t أو نوع ssize_t.

يقوم الرأس <regex.h> بتعريف نوع الهيكل regmatch_t، والذي يجب أن يتضمن على الأقل الأعضاء التالية:

regoff_t    rm_so    إزاحة البايت من بداية السلسلة
                     إلى بداية السلسلة الفرعية.
regoff_t    rm_eo    إزاحة البايت من بداية السلسلة إلى
                     الحرف الأول بعد نهاية السلسلة الفرعية.

يقوم الرأس <regex.h> بتعريف الثوابت الرمزية التالية لمعامل cflags للدالة regcomp():

REG_EXTENDED
استخدم تعبيرات عادية موسعة.
REG_ICASE
تجاهل حالة الأحرف في المطابقة.
REG_NOSUB
الإبلاغ فقط عن النجاح أو الفشل في regexec().
REG_NEWLINE
تغيير التعامل مع <السطر الجديد>.

يقوم الرأس <regex.h> بتعريف الثوابت الرمزية التالية لمعامل eflags للدالة regexec():

REG_NOTBOL
الحرف <قبعة> ('^')، عند اعتباره حرفًا خاصًا، لا يطابق بداية string.
REG_NOTEOL
الحرف <علامة الدولار> ('$')، عند اعتباره حرفًا خاصًا، لا يطابق نهاية string.

يقوم الرأس <regex.h> بتعريف الثوابت الرمزية التالية كقيم إرجاع للخطأ:

REG_NOMATCH
فشلت regexec() في المطابقة.
REG_BADPAT
تعبير عادي غير صالح.
REG_ECOLLATE
تم الرجوع إلى عنصر تجميع غير صالح.
REG_ECTYPE
تم الرجوع إلى نوع فئة أحرف غير صالح.
REG_EESCAPE
حرف <شرطة مائلة عكسية> لاحق في النمط.
REG_ESUBREG
الرقم في \digit غير صالح أو في خطأ.
REG_EBRACK
عدم توازن "[]".
REG_EPAREN
عدم توازن "\(\)" أو "()".
REG_EBRACE
عدم توازن "\{\}".
REG_BADBR
محتوى "\{\}" غير صالح: ليس رقمًا، الرقم كبير جدًا، أكثر من رقمين، الأول أكبر من الثاني.
REG_ERANGE
نقطة نهاية غير صالحة في تعبير النطاق.
REG_ESPACE
نفاد الذاكرة.
REG_BADRPT
'?'، '*'، أو '+' لا يسبقه تعبير عادي صالح.

يجب تعريف ما يلي كدوال وقد يتم تعريفه أيضًا كـ macros. يجب توفير تعريفات الوظائف.

int    regcomp(regex_t *restrict, const char *restrict, int);
size_t regerror(int, const regex_t *restrict, char *restrict, size_t);
int    regexec(const regex_t *restrict, const char *restrict, size_t,
           regmatch_t [restrict], int);
void   regfree(regex_t *);

قد يقوم التنفيذ بتعريف macros أو ثوابت إضافية باستخدام أسماء تبدأ بـ REG_.

الاسم

regex.h - مواصفات التعبيرات النمطية POSIX

الملخص

#include <regex.h>

الوصف

ملف الرأس <regex.h> يعرّف الثوابت والمتغيرات والدوال الخاصة بمعالجة التعبيرات النمطية.

يتم استخدام التعبيرات النمطية لمطابقة الأنماط في السلاسل النصية.

قيمة الإرجاع

تُرجع الدوال ذات الصلة أعدادًا صحيحة للإشارة إلى النجاح أو الفشل.

الأخطاء

تُرجع الدوال أعدادًا صحيحة غير صفرية للإشارة إلى حدوث أخطاء.

متوافق مع

POSIX.1-2001, POSIX.1-2008.

ملاحظات

يجب تضمين ملف الرأس <regex.h> في أي برنامج يستخدم وظائف التعبيرات النمطية.

مثال

#include <stdio.h>
#include <regex.h>

int main() {
    regex_t regex;
    int reti;
    char msgbuf[100];

    /* تجميع التعبير النمطي */
    reti = regcomp(&regex, "^a[b-d]*c$", REG_EXTENDED);
    if (reti) {
        fprintf(stderr, "Could not compile regex\n");
        return 1;
    }

    /* تنفيذ المطابقة */
    reti = regexec(&regex, "abc", 0, NULL, 0);
    if (!reti) {
        puts("Match!");
    } else if (reti == REG_NOMATCH) {
        puts("No match.");
    } else {
        regerror(reti, &regex, msgbuf, sizeof(msgbuf));
        fprintf(stderr, "Regex match failed: %s\n", msgbuf);
    }

    /* تحرير الذاكرة */
    regfree(&regex);

    return 0;
}
