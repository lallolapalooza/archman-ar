rmpc(1)

الحزمة: rmpc

التراخيص: BSD-3-Clause

rmpc(1) دليل الأوامر العامة rmpc(1)

الاسم

rmpc

الملخص

rmpc [-c|--config] [-t|--theme] [-a|--address] [-p|--password] [--partition] [--autocreate] [-h|--help] [أوامر فرعية]

الوصف

خيارات

-c, --config=ملف
-t, --theme=ملف
-a, --address=عنوان
تجاوز العنوان للاتصال به. الافتراضي هو القيمة الموجودة في ملف الإعدادات
-p, --password=كلمة المرور
تجاوز كلمة مرور MPD
--partition=قسم
القسم للاتصال به عند بدء التشغيل
--autocreate
إنشاء القسم تلقائيًا إذا لم يكن موجودًا. يتطلب تعيين القسم
-h, --help
طباعة المساعدة

أوامر فرعية

rmpc-addrandom(1)
rmpc-config(1)
يطبع الإعدادات الافتراضية. يمكن استخدامه لتهيئة ملف الإعدادات الخاص بك
rmpc-theme(1)
يطبع السمات الافتراضية. يمكن استخدامه لتهيئة ملف السمات الخاص بك
rmpc-lyricsindex(1)
فهرسة دليل كلمات الأغاني وعرض النتيجة، مخصص لأغراض التصحيح فقط
rmpc-update(1)
مسح دليل موسيقى MPD بحثًا عن تحديثات
rmpc-rescan(1)
مسح دليل موسيقى MPD بحثًا عن تحديثات. يعيد مسح الملفات غير المعدلة أيضًا
rmpc-albumart(1)
يحفظ فن الألبوم الحالي في ملف. رموز الخروج: * 0: نجاح * 1: خطأ * 2: لم يتم العثور على فن ألبوم * 3: لا توجد أغنية قيد التشغيل
rmpc-debuginfo(1)
يطبع معلومات حول تبعيات وقت التشغيل الاختيارية
rmpc-version(1)
يطبع إصدار rmpc
rmpc-queue(1)
يطبع قائمة الأغاني في قائمة التشغيل الحالية
rmpc-listall(1)
يسرد الملفات من قاعدة بيانات الموسيقى. إذا لم يتم تحديد ملف، يسرد جميع الأغاني في قاعدة البيانات
rmpc-play(1)
يشغل الأغنية في الموضع المحدد في قائمة التشغيل الحالية. الافتراضي هو الأغنية المتوقفة مؤقتًا حاليًا
rmpc-pause(1)
إيقاف التشغيل مؤقتًا
rmpc-unpause(1)
استئناف التشغيل
rmpc-togglepause(1)
يبدل بين التشغيل والإيقاف المؤقت
rmpc-stop(1)
إيقاف التشغيل
rmpc-next(1)
يشغل الأغنية التالية في قائمة التشغيل
rmpc-prev(1)
يشغل الأغنية السابقة في قائمة التشغيل
rmpc-volume(1)
يضبط مستوى الصوت، نسبيًا إذا كان مسبوقًا بـ + أو -. يطبع مستوى الصوت الحالي إذا لم يتم إعطاء وسائط
rmpc-repeat(1)
تشغيل أو إيقاف
rmpc-random(1)
تشغيل أو إيقاف
rmpc-single(1)
تشغيل، إيقاف، أو تشغيل مرة واحدة
rmpc-consume(1)
تشغيل، إيقاف، أو تشغيل مرة واحدة
rmpc-togglerepeat(1)
يبدل وضع التكرار
rmpc-togglerandom(1)
يبدل وضع العشوائية
rmpc-togglesingle(1)
يبدل وضع المفرد
rmpc-toggleconsume(1)
يبدل وضع الاستهلاك
rmpc-seek(1)
يقفز في الأغنية الحالية (بالثواني)، نسبيًا إذا كان مسبوقًا بـ + أو -
rmpc-clear(1)
مسح قائمة التشغيل الحالية
rmpc-add(1)
إضافة أغنية إلى قائمة التشغيل الحالية. نسبيًا إلى جذر قاعدة بيانات الموسيقى. '/' لإضافة جميع الملفات إلى قائمة التشغيل
rmpc-addyt(1)
إضافة أغنية من يوتيوب إلى قائمة التشغيل الحالية
rmpc-searchyt(1)
البحث وإضافة أول نتيجة مطابقة إلى قائمة التشغيل الحالية (يوتيوب افتراضيًا)
rmpc-outputs(1)
يسرد مخرجات MPD
rmpc-toggleoutput(1)
تبديل مخرج MPD تشغيل أو إيقاف
rmpc-enableoutput(1)
تمكين مخرج MPD
rmpc-disableoutput(1)
تعطيل مخرج MPD
rmpc-decoders(1)
يسرد إضافات فك ترميز MPD
rmpc-status(1)
يطبع معلومات مختلفة مثل حالة التشغيل
rmpc-song(1)
يطبع معلومات حول الأغنية الحالية. إذا تم تحديد --path، يطبع معلومات حول الأغنية في المسار المحدد بدلاً من ذلك. إذا تم تحديد --path عدة مرات، يطبع مصفوفة تحتوي على جميع الأغاني
rmpc-mount(1)
يربط وسائط التخزين المدعومة بـ MPD
rmpc-unmount(1)
يفصل وسيط التخزين بالاسم المحدد
rmpc-listmounts(1)
يسرد وسائط التخزين المرتبطة حاليًا
rmpc-listpartitions(1)
يسرد الأقسام الموجودة حاليًا
rmpc-sticker(1)
يتعامل مع ملصقات الأغاني ويستعلم عنها
rmpc-remote(1)
يرسل أمرًا عن بعد إلى مثيل rmpc قيد التشغيل
rmpc-sendmessage(1)
يرسل رسالة إلى القناة المحددة للتواصل بين العملاء
rmpc-help(1)
اطبع هذه الرسالة أو المساعدة للأوامر الفرعية المحددة
