struct send_data {
  uint8_t      code;          // رمز حزمة RADIUS. 
  uint8_t      seq_nbr;       // رقم تسلسل الحزمة. 
  char        *server;        // اسم/عنوان خادم RADIUS. 
  int          svc_port;      // منفذ وجهة بروتوكول RADIUS. 
  char        *secret;        // السر المشترك لخادم RADIUS. 
  int          timeout;       // مهلة الجلسة بالثواني. 
  int          retries;
  VALUE_PAIR  *send_pairs;    // أزواج قيم/سمات إضافية للإرسال. 
  VALUE_PAIR  *receive_pairs; // المكان الذي سيتم فيه وضع أزواج القيم/السمات المستلمة. 
};
