rc_avpair_tostr(3)

الحزمة: radcli

التراخيص: BSD

rc_avpair_tostr(3) مكتبة عميل Radius rc_avpair_tostr(3)

الاسم

rc_avpair_tostr -

المُلَخَّص

#include <radcli/radcli.h>

int rc_avpair_tostr(
    rc_handle const  *rh,
    VALUE_PAIR       *pair,
    char             *name,
    int               ln,
    char             *value,
    int               lv
);

الوصف

ترجمة زوج av_pair إلى سلاسل قابلة للطباعة

المُعاملات

rh
مقبض للإعدادات المحللة.
pair
مؤشر إلى بنية VALUE_PAIR.
name
اسم الزوج.
ln
حجم الاسم.
value
قيمة الزوج.
lv
حجم القيمة.

البُنى

تجنب استخدام هذه البنية مباشرة. استخدم وظائف rc_avpair_get_.


struct rc_value_pair {
  char                   name[RC_NAME_LENGTH+1];      // اسم السمة إذا كان معروفًا. 
  uint64_t               attribute; // قيمة رقمية للسمة من نوع rc_attr_id بما في ذلك البائع؛ استخدم VENDOR() و ATTRID() للفصل. 
  rc_attr_type           type;      // نوع السمة. 
  uint32_t               lvalue;    // قيمة السمة إذا كان النوع PW_TYPE_INTEGER أو PW_TYPE_DATE أو PW_TYPE_IPADDR. 
  char                   strvalue[AUTH_STRING_LEN+1];  // تحتوي على قيمة السمة في حالات أخرى. 
  struct rc_value_pair  *next;
  char                   pad[32];       // حشو غير مستخدم 
};

القيمة المُعادَة

0 عند النجاح، -1 عند الفشل.
