restic-key(1)

الحزمة: restic

التراخيص: BSD

restic backup(1) restic backup(1)

الاسم

restic-key - إدارة المفاتيح (كلمات المرور)

ملخص

restic key [علامات]

الوصف

يسمح لك أمر "key" بتعيين مفاتيح وصول متعددة أو كلمات مرور لكل مستودع.

خيارات

-h, --help[=false] مساعدة لأمر key

الخيارات الموروثة من الأوامر الأصل

--cacert=[] ملف لتحميل شهادات الجذر منه (الافتراضي: استخدام شهادات النظام أو $RESTIC_CACERT)

--cache-dir="" تعيين دليل ذاكرة التخزين المؤقت. (الافتراضي: استخدام دليل ذاكرة التخزين المؤقت الافتراضي للنظام)

--cleanup-cache[=false] إزالة مجلدات ذاكرة التخزين المؤقت القديمة تلقائيًا

--compression=auto وضع الضغط (متاح فقط لإصدار تنسيق المستودع 2)، أحد الخيارات (auto|off|max) (الافتراضي: $RESTIC_COMPRESSION)

--http-user-agent="" تعيين وكيل مستخدم HTTP لطلبات HTTP الصادرة

--insecure-no-password[=false] استخدام كلمة مرور فارغة للمستودع، يجب تمريرها لكل أمر restic (غير آمن)

--insecure-tls[=false] تخطي التحقق من شهادة TLS عند الاتصال بالمستودع (غير آمن)

--json[=false] تعيين وضع الإخراج إلى JSON للأوامر التي تدعمه

--key-hint="" معرف المفتاح للمفتاح لمحاولة فك تشفيره أولاً (الافتراضي: $RESTIC_KEY_HINT)

--limit-download=0 يحد التنزيلات بحد أقصى معدل بالكيلوبايت/ثانية. (الافتراضي: غير محدود)

--limit-upload=0 يحد التحميلات بحد أقصى معدل بالكيلوبايت/ثانية. (الافتراضي: غير محدود)

--no-cache[=false] عدم استخدام ذاكرة تخزين مؤقت محلية

--no-extra-verify[=false] تخطي التحقق الإضافي من البيانات قبل التحميل (انظر التوثيق)

--no-lock[=false] عدم قفل المستودع، هذا يسمح ببعض العمليات على المستودعات للقراءة فقط

-o, --option=[] تعيين خيار ممتد (key=value، يمكن تحديده عدة مرات)

--pack-size=0 تعيين حجم الحزمة المستهدف بالميغابايت، قد تكون ملفات الحزمة التي تم إنشاؤها أكبر (الافتراضي: $RESTIC_PACK_SIZE)

--password-command="" أمر shell للحصول على كلمة مرور المستودع من (الافتراضي: $RESTIC_PASSWORD_COMMAND)

-p, --password-file="" ملف للقراءة منه كلمة مرور المستودع (الافتراضي: $RESTIC_PASSWORD_FILE)

-q, --quiet[=false] عدم إخراج تقرير تقدم شامل

-r, --repo="" المستودع للنسخ الاحتياطي إليه أو الاستعادة منه (الافتراضي: $RESTIC_REPOSITORY)

--repository-file="" ملف للقراءة منه موقع المستودع (الافتراضي: $RESTIC_REPOSITORY_FILE)

--retry-lock=0s إعادة المحاولة لقفل المستودع إذا كان مقفلاً بالفعل، يأخذ قيمة مثل 5m أو 2h (الافتراضي: لا توجد محاولات إعادة)

--stuck-request-timeout=5m0s مدة سيتم بعدها إعادة المحاولة في الطلبات العالقة

--tls-client-cert="" مسار إلى ملف يحتوي على شهادة عميل TLS مشفرة بصيغة PEM ومفتاح خاص (الافتراضي: $RESTIC_TLS_CLIENT_CERT)

-v, --verbose[=0] كن مطولاً (حدد عدة مرات أو مستوى باستخدام --verbose=n، الحد الأقصى للمستوى/المرات هو 2)
