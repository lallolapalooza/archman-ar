rebuilderd-worker.conf(5) دليل تنسيقات الملفات rebuilderd-worker.conf(5)

الاسم

rebuilderd-worker.conf - ملفات إعداد rebuilderd-worker

ملخص

/etc/rebuilderd-worker.conf

الوصف

يقوم هذا الملف بتهيئة rebuilder-worker connect ويحدد نقطة النهاية وسر التسجيل (signup_secret). كلاهما اختياري ولكنه يتطلب تحديد نقطة النهاية بشكل صريح عبر سطر الأوامر إذا لم يتم تكوينها في الإعداد.

الخيارات

endpoint=

rebuilderd للاتصال به.

signup_secret=

سيسمح الخادم بإدراج مفتاحنا في القائمة البيضاء أو يتطلب سر تسجيل.

[build]

timeout=

اضبط مهلة بالثواني يتم بعدها إنهاء diffoscope (افتراضيًا 24 ساعة).

max_bytes=

اضبط حدًا أقصى لحجم إخراج diffoscope بالبايت (افتراضي: لا شيء). عند الوصول إلى هذا الحد، يتم اقتطاع السجل ولكن لن يتم إنهاء واجهة خلفية rebuilder.

silent=

بشكل افتراضي، يتم توجيه إخراج البناء إلى stdout/stderr. يمكن تعطيل ذلك عن طريق تعيين هذا إلى true.

[diffoscope]

enabled=

إنشاء وإرفاق فروقات باستخدام diffoscope عند إعادة البناء.

args=

تمرير وسيطات إضافية إلى diffoscope. استخدم بحكمة، فقد لا تعمل بعض الخيارات بشكل جيد.

timeout=

اضبط مهلة بالثواني يتم بعدها إنهاء diffoscope (افتراضيًا ساعة واحدة).

max_bytes=

اضبط حدًا أقصى لحجم إخراج diffoscope بالبايت (افتراضي: لا شيء). عند الوصول إلى هذا الحد، يتم إنهاء diffoscope ويتم اقتطاع الإخراج.

مثال

## The rebuilderd to connect to
endpoint = "http://127.0.0.1:8484"
## The server would either allowlist our key or require a signup secret
#signup_secret = "your_signup_key"
[build]
#timeout = 86400 # 24 hours
## Set a maximum build log limit in bytes (default: none).
## When reaching this limit the log is truncated but the rebuilder backend is *not* terminated.
max_bytes = 10485760 # 10 MiB
[diffoscope]
## Generate and attach diffs with diffoscope when rebuilding
enabled = false
## Pass additional arguments to diffoscope. Use wisely, some options might not work well.
#args = ["--max-container-depth", "2", "--fuzzy-threshold", "0"]
## Set a timeout in seconds after which diffoscope is terminated (default: 3600)
#timeout = 600 # 10 minutes
## Set a maximum diffoscope output limit in bytes (default: none).
## When reaching this limit diffoscope is terminated and the output is truncated.
max_bytes = 41943040 # 40 MiB
