recv(2) دليل استدعاءات النظام recv(2)

الاسم

recv, recvfrom, recvmsg - استقبال رسالة من مقبس

المكتبة

مكتبة C القياسية (libc, -lc)

ملخص

#include <sys/socket.h>
ssize_t recv(size_t size;
                 int sockfd, void buf[size], size_t size,
                 int flags);
ssize_t recvfrom(size_t size;
                 int sockfd, void buf[restrict size], size_t size,
                 int flags,
                 struct sockaddr *_Nullable restrict src_addr,
                 socklen_t *_Nullable restrict addrlen);
ssize_t recvmsg(int sockfd, struct msghdr *msg, int flags);

الوصف

تُستخدم استدعاءات recv() و recvfrom() و recvmsg() لاستقبال الرسائل من مقبس. يمكن استخدامها لاستقبال البيانات على المقابس المتصلة وغير المتصلة. تشرح هذه الصفحة أولاً الميزات الشائعة لجميع استدعاءات النظام الثلاثة، ثم تصف الاختلافات بين الاستدعاءات.

الاختلاف الوحيد بين recv() و read(2) هو وجود flags. مع قيمة صفر لوسيط flags، فإن recv() مكافئة بشكل عام لـ read(2) (لكن انظر الملاحظات). أيضًا، الاستدعاء التالي


recv(sockfd, buf, size, flags);

مكافئ لـ


recvfrom(sockfd, buf, size, flags, NULL, NULL);

ترجع جميع الاستدعاءات الثلاثة حجم الرسالة عند الاكتمال بنجاح. إذا كانت الرسالة طويلة جدًا بحيث لا تتسع في المخزن المؤقت المقدم، فقد يتم تجاهل البايتات الزائدة اعتمادًا على نوع المقبس الذي تم استلام الرسالة منه.

إذا لم تكن هناك رسائل متاحة على المقبس، تنتظر استدعاءات الاستقبال وصول رسالة، إلا إذا كان المقبس غير مانع (راجع fcntl(2))، وفي هذه الحالة يتم إرجاع القيمة -1 ويتم تعيين errno إلى EAGAIN أو EWOULDBLOCK. ترجع استدعاءات الاستقبال عادةً أي بيانات متاحة، حتى الكمية المطلوبة، بدلاً من انتظار استلام الكمية الكاملة المطلوبة.

يمكن للتطبيق استخدام select(2) أو poll(2) أو epoll(7) لتحديد متى تصل المزيد من البيانات على مقبس.

الوسيط flags

يتم تشكيل الوسيط flags عن طريق ORing قيمة واحدة أو أكثر من القيم التالية:

MSG_CMSG_CLOEXEC (recvmsg() فقط؛ منذ Linux 2.6.23)
تعيين علامة الإغلاق عند التنفيذ لوصف الملف الذي تم استلامه عبر وصف ملف نطاق UNIX باستخدام العملية SCM_RIGHTS (الموضحة في unix(7)). هذه العلامة مفيدة لنفس أسباب علامة O_CLOEXEC لـ open(2).
MSG_DONTWAIT (منذ Linux 2.2)
تمكين التشغيل غير المانع؛ إذا كانت العملية ستؤدي إلى الحظر، يفشل الاستدعاء مع EAGAIN أو EWOULDBLOCK. يوفر هذا سلوكًا مشابهًا لتعيين علامة O_NONBLOCK (عبر العملية F_SETFL لـ fcntl(2))، لكنه يختلف في أن MSG_DONTWAIT هي خيار لكل استدعاء، بينما O_NONBLOCK هو إعداد على وصف الملف المفتوح (راجع open(2))، والذي سيؤثر على جميع الخيوط في العملية المتصلة بالإضافة إلى العمليات الأخرى التي تحمل واصفات ملف تشير إلى نفس وصف الملف المفتوح.
MSG_ERRQUEUE (منذ Linux 2.2)
تحدد هذه العلامة أن الأخطاء المعلقة يجب استقبالها من طابور أخطاء المقبس. يتم تمرير الخطأ في رسالة ملحقة بنوع يعتمد على البروتوكول (لـ IPv4 IP_RECVERR). يجب على المستخدم توفير مخزن مؤقت بحجم كافٍ. راجع cmsg(3) و ip(7) لمزيد من المعلومات. يتم تمرير حمولة الحزمة الأصلية التي تسببت في الخطأ كبيانات عادية عبر msg_iovec. يتم توفير عنوان الوجهة الأصلي للحزمة التي تسببت في الخطأ عبر msg_name.
يتم توفير الخطأ في بنية sock_extended_err:

#define SO_EE_ORIGIN_NONE    0
#define SO_EE_ORIGIN_LOCAL   1
#define SO_EE_ORIGIN_ICMP    2
#define SO_EE_ORIGIN_ICMP6   3
struct sock_extended_err
{
    uint32_t ee_errno;   /* رقم الخطأ */
    uint8_t  ee_origin;  /* مصدر الخطأ */
    uint8_t  ee_type;    /* النوع */
    uint8_t  ee_code;    /* الرمز */
    uint8_t  ee_pad;     /* حشو */
    uint32_t ee_info;    /* معلومات إضافية */
    uint32_t ee_data;    /* بيانات أخرى */
    /* قد تتبعها بيانات أخرى */
};
struct sockaddr *SO_EE_OFFENDER(struct sock_extended_err *);

يحتوي ee_errno على رقم errno للخطأ المعلق. ee_origin هو رمز مصدر الخطأ. الحقول الأخرى خاصة بالبروتوكول. الماكرو SO_EE_OFFENDER يرجع مؤشرًا إلى عنوان الكائن الشبكي الذي نشأ منه الخطأ مع إعطاء مؤشر للرسالة الملحقة. إذا كان هذا العنوان غير معروف، فإن العضو sa_family لـ sockaddr يحتوي على AF_UNSPEC والحقول الأخرى لـ sockaddr غير محددة. حمولة الحزمة التي تسببت في الخطأ يتم تمريرها كبيانات عادية.
بالنسبة للأخطاء المحلية، لا يتم تمرير عنوان (يمكن التحقق من ذلك باستخدام العضو cmsg_len لـ cmsghdr). بالنسبة لاستقبال الأخطاء، يتم تعيين علامة MSG_ERRQUEUE في msghdr. بعد تمرير خطأ، يتم إعادة إنشاء الخطأ المعلق بناءً على الخطأ المعلق التالي وسيتم تمريره عند عملية المقبس التالية.
MSG_OOB
تطلب هذه العلامة استقبال بيانات خارج النطاق (out-of-band) التي لن يتم استقبالها في تدفق البيانات العادي. تضع بعض البروتوكولات البيانات المعجلة في بداية طابور البيانات العادي، وبالتالي لا يمكن استخدام هذه العلامة مع مثل هذه البروتوكولات.
MSG_PEEK
تتسبب هذه العلامة في أن تعيد عملية الاستقبال البيانات من بداية طابور الاستقبال دون إزالة تلك البيانات من الطابور. وبالتالي، ستقوم عملية استقبال لاحقة بإرجاع نفس البيانات.
MSG_TRUNC (منذ Linux 2.2)
لمقابس raw (AF_PACKET) و datagram للإنترنت (منذ Linux 2.4.27/2.6.8) و netlink (منذ Linux 2.6.22) و UNIX datagram و sequenced-packet (منذ Linux 3.4): قم بإرجاع الحجم الفعلي للحزمة أو datagram، حتى عندما تكون أطول من المخزن المؤقت المقدم.
للاستخدام مع مقابس تدفق الإنترنت، انظر tcp(7).
MSG_WAITALL (منذ Linux 2.2)
تطلب هذه العلامة أن تمنع العملية حتى يتم تلبية الطلب بالكامل. ومع ذلك، قد ترجع الاستدعاء بيانات أقل مما هو مطلوب إذا تم التقاط إشارة، أو حدث خطأ أو انقطاع، أو كانت البيانات التالية التي سيتم استقبالها من نوع مختلف عن الذي تم إرجاعه. هذه العلامة ليس لها تأثير على مقابس datagram.

recvfrom()

تضع recvfrom() الرسالة المستقبلة في المخزن المؤقت buf. يجب على المتصل تحديد حجم المخزن المؤقت في size.

إذا لم يكن src_addr NULL، وإذا كان البروتوكول الأساسي يوفر عنوان المصدر للرسالة، فسيتم وضع هذا العنوان المصدر في المخزن المؤقت الذي يشير إليه src_addr. في هذه الحالة، addrlen هو وسيط قيمة-نتيجة. قبل الاستدعاء، يجب تهيئته إلى حجم المخزن المؤقت المرتبط بـ src_addr. عند العودة، يتم تحديث addrlen ليحتوي على الحجم الفعلي لعنوان المصدر. يتم اقتطاع العنوان المرجع إذا كان المخزن المؤقت المقدم صغيرًا جدًا؛ في هذه الحالة، سيعيد addrlen قيمة أكبر مما تم توفيره للاستدعاء.

إذا لم يكن المتصل مهتمًا بعنوان المصدر، فيجب تحديد src_addr و addrlen كـ NULL.
